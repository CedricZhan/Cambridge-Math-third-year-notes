\section*{Principles of Statistics}

\subsection*{0. INTRODUCTION}

Let $(\Omega,\mathcal{F},\mathbb{P})$ be a probability space, and
$X:\Omega\to\mathbb{R}$ a random variable with distribution function
\[
\begin{aligned}
F(t)
&=
\mathbb{P}\bigl(\{\omega:X(\omega)\leq t\}\bigr) \\
&=
\begin{cases}
\displaystyle \int_{-\infty}^{t} f(x)\,dx,
& \text{if $X$ is continuous with pdf $f$}, \\[1ex]
\displaystyle \sum_{x\leq t} f(x),
& \text{if $X$ is discrete with pmf $f$}.
\end{cases}
\end{aligned}
\]

We denote $P=\mathbb{P}\circ X^{-1}$, the law or ``distribution'' of $X$.

We observe $X_1,\ldots,X_n$, which are iid copies of $X$, where $n$ is
the \underline{sample size}.

\paragraph{Def 1.}
A \underline{statistical model} is a collection
$\{f_\theta:\theta\in\Theta\}$ of pdfs or pmfs, or
$\{P_\theta:\theta\in\Theta\}$ of laws, indexed by a
\underline{parameter space} $\Theta$.

\paragraph{Ex.}
\begin{itemize}
    \item $N(\theta,1)$, \qquad $\Theta=\mathbb{R}$.
    \item $N(\mu,\sigma^2)$, \qquad
    $\theta=(\mu,\sigma^2)$, \quad
    $\Theta=\mathbb{R}\times(0,\infty)$.
    \item $\operatorname{Poi}(\theta)$, \qquad
    $\Theta=[0,\infty)$.
\end{itemize}

\paragraph{Def 2.}
We say the model is \underline{correctly specified} when
$\exists\,\theta_0\in\Theta$ such that $P=P_{\theta_0}$.

We will refer to the ``true parameter'' as $\theta_0$.

\subsection*{Goals of Statistics}

\begin{enumerate}
    \item \underline{Estimation}:
    We define an \underline{estimator}
    \[
    \widehat{\theta}_n
    =
    \widehat{\theta}_n(X_1,\ldots,X_n),
    \]
    which we hope is close to $\theta_0$ when
    $X_i\overset{\mathrm{iid}}{\sim}P_{\theta_0}$,
    for all $\theta_0\in\Theta$, when $n$ is large.

    \item \underline{Testing}:
    We have a null hypothesis
    \[
    H_0:\theta\in\Theta_0\subseteq\Theta,
    \]
    and an alternative hypothesis
    \[
    H_1:\theta\in\Theta_1\subseteq\Theta.
    \]
    We define a \underline{test},
    \[
    \psi_n:=\psi_n(X_1,\ldots,X_n),
    \]
    such that $\psi_n=0$ if $H_0$ is true, and
    $\psi_n=1$ if $H_1$ is true, \underline{w.h.p.}

    \item \underline{Uncertainty Quantification}
    \begin{itemize}
        \item Define a \underline{confidence set}
        $C_n(X_1,\ldots,X_n;\alpha)$, such that
        \[
        P_\theta(\theta\in C_n)
        \overset{(\geq)}{=}
        1-\alpha
        \qquad \forall\,\theta\in\Theta,
        \]
        where $\alpha$ is the \underline{significance level}.

        \item Define a \underline{Bayesian credible set},
        a region of high posterior probability.
    \end{itemize}
\end{enumerate}

\section*{I. LIKELIHOOD PRINCIPLE}

\paragraph{Def 3.}
The \underline{likelihood function}
\[
L_n(\theta)=\prod_{i=1}^{n}f_\theta(X_i).
\]

\noindent\underline{Note:}
This is a random function, as it depends implicitly on $X_1,\ldots,X_n$.

The \underline{log-likelihood function}
\[
\ell_n(\theta)=\sum_{i=1}^{n}\log f_\theta(X_i),
\]
we write
\[
\overline{\ell}_n(\theta)=\frac{1}{n}\ell_n(\theta)
\]
for a \underline{normalised log-likelihood}.

\paragraph{Def 4.}
A \underline{maximum likelihood estimator}
$\widehat{\theta}_{\mathrm{MLE}}$ is any estimator with
\[
L_n\bigl(\widehat{\theta}_{\mathrm{MLE}}\bigr)
=
\max_{\theta\in\Theta}L_n(\theta).
\]

\paragraph{Ex.}
\[
X_i\overset{\mathrm{iid}}{\sim}\operatorname{Poi}(\theta).
\]
\[
L_n(\theta)
=
\prod_{i=1}^{n}e^{-\theta}\frac{\theta^{X_i}}{X_i!}.
\]
\[
\ell_n(\theta)
=
-n\theta+(\log\theta)\sum X_i+\mathrm{const.}
\]
\[
\ell_n'(\theta)
=
-n+\frac{\sum X_i}{\theta}
\overset{\mathrm{FOC}}{=}0
\]
\[
\Longrightarrow\quad
\text{suggests }
\widehat{\theta}_{\mathrm{MLE}}
=
\frac{\displaystyle\sum_{i=1}^{n}X_i}{n}
:=
\overline{X}_n
\qquad
\text{``sample mean''}.
\]

\noindent\underline{SOC:}
\[
\ell_n''(\theta)
=
-\frac{1}{\theta^2}\sum X_i
<0
\qquad
\text{if }\sum X_i>0.
\]

If $\sum X_i=0$,
\[
\ell_n(\theta)=-n\theta
\quad\Longrightarrow\quad
\widehat{\theta}_{\mathrm{MLE}}=0.
\]

In every case
\[
\widehat{\theta}_{\mathrm{MLE}}=\overline{X}_n.
\]

\paragraph{Ex.}
\[
X_i\overset{\mathrm{iid}}{\sim}N(\mu,\sigma^2),
\qquad
\theta=(\mu,\sigma^2).
\]
\[
\ell_n(\theta)
=
-\frac{n}{2}\log\pi
-\frac{n}{2}\log(\sigma^2)
-\frac{1}{2\sigma^2}
\sum_{i=1}^{n}(X_i-\mu)^2.
\]

\noindent\underline{FOC:}
\[
\frac{\partial\ell_n}{\partial\mu}
=
-\frac{1}{\sigma^2}
\sum_{i=1}^{n}(X_i-\mu)
=0,
\]
\[
\frac{\partial\ell_n}{\partial\sigma^2}
=
-\frac{n}{2\sigma^2}
+
\frac{1}{2(\sigma^2)^2}
\sum_{i=1}^{n}(X_i-\mu)^2
=0.
\]

We can check that the Hessian of $\ell_n$ is negative definite
$\forall\,\theta\in\mathbb{R}\times(0,\infty)$.

Hence,
\[
\widehat{\theta}_{\mathrm{MLE}}
=
\begin{pmatrix}
\overline{X}_n \\[6pt]
\underbrace{
\displaystyle\frac{1}{n}
\sum_{i=1}^{n}(X_i-\overline{X}_n)^2
}_{\text{``Sample variance''}}
\end{pmatrix}.
\]

\subsection*{Non-examinable}

The likelihood principle poses that if two observations
\[
X=(X_1,\ldots,X_n)
\qquad\text{and}\qquad
X^*=(X_1^*,\ldots,X_n^*)
\]
give rise to likelihoods $L_n,L_n^*$ respectively, such that
\[
L_n(\theta)=L_n^*(\theta)\cdot M
\qquad \forall\,\theta\in\Theta,
\]
and some constant $M$, then they should produce the same
``inference'' on the true parameter.

\begin{itemize}
    \item The MLE, Bayes estimators, \underline{credible sets}
    satisfy the likelihood principle.
    \item \underline{Confidence sets}, tests generally do not.
\end{itemize}

\noindent\underline{Birnbaum (1962)}
\[
\text{Conditionality}+\text{Sufficiency}
\quad\Longrightarrow\quad
\text{Likelihood principle}.
\]

\section*{II. Information Geometry}

The \underline{score function} is
\[
S_n(\theta)
=
\frac{\partial}{\partial\theta}\ell_n(\theta),
\qquad
\Theta\subseteq\mathbb{R}^d.
\]

In many cases,
\[
S_n(\widehat{\theta}_{\mathrm{MLE}})=0,
\]
where, by convention,
\[
S_n(\theta')
=
\left.
\frac{\partial}{\partial\theta}\ell_n(\theta)
\right|_{\theta=\theta'}.
\]

Let $X$ be a r.v.\ taking values in $\mathcal{X}$, and
$g:\mathcal{X}\to\mathbb{R}$.
Take a model $\{f(x,\theta):\theta\in\Theta\}$.
We will write
\[
\mathbb{E}_{\theta}g(X)
=
\int g(x)f(x,\theta)\,dx,
\qquad
X\sim f(\cdot,\theta).
\]

\medskip
\noindent\textbf{Obs 1.}

Suppose the model $\{f(\cdot,\theta):\theta\in\Theta\}$ is such that
\[
\mathbb{E}_{\theta_0}\bigl|\log f(X,\theta)\bigr|
<
\infty
\qquad
\forall\theta\in\Theta.
\]
Define
\[
\ell(\theta)
=
\mathbb{E}_{\theta_0}\log f(X,\theta),
\qquad
X\sim P_{\theta_0}.
\]

The normalised log-likelihood
\[
\overline{\ell}_n(\theta)
=
\frac{1}{n}\sum_{i=1}^{n}\log f(X_i,\theta)
\]
can be thought of as a sample version of $\ell(\theta)$.

Suppose the model is correctly specified, such that $X$ has pdf
$f(\cdot,\theta_0)$ for some $\theta_0\in\Theta$.

\medskip
\noindent\textbf{Fact.}
Then $\ell(\theta)$ is maximised at $\theta_0$.
This justifies using the MLE to approximate $\theta_0$ when $n$ is large.

Indeed,
\[
\begin{aligned}
\ell(\theta)-\ell(\theta_0)
&=
\mathbb{E}_{\theta_0}
\left(
\log\frac{f(X,\theta)}{f(X,\theta_0)}
\right)
\\
&\leq
\log\mathbb{E}_{\theta_0}
\left(
\frac{f(X,\theta)}{f(X,\theta_0)}
\right)
\\
&=
\log\int
\frac{f(x,\theta)}{f(x,\theta_0)}
f(x,\theta_0)\,dx
\\
&=
\log 1=0.
\end{aligned}
\]
By Jensen's inequality, as $\log$ is concave, with equality if
\[
\frac{f(X,\theta)}{f(X,\theta_0)}
\]
is a.s.\ constant.

\[
\therefore\quad
\ell(\theta)\leq\ell(\theta_0).
\]

If
\[
f(\cdot,\theta)=f(\cdot,\theta')
\quad\Longrightarrow\quad
\theta=\theta',
\]
we say the parameter is \underline{identifiable}.

In that case, if $\theta\neq\theta_0$, Jensen's inequality is strict,
so $\theta_0$ is actually the unique maximizer of $\ell$.

\medskip
\noindent\textbf{Rmk.}
The difference
\[
\ell(\theta_0)-\ell(\theta)
=
\int
\log\frac{f(x,\theta_0)}{f(x,\theta)}
f(x,\theta_0)\,dx
\]
is called the \underline{Kullback--Leibler divergence}
or \underline{relative entropy} between $P_{\theta_0}$ and $P_\theta$.

A notion of distance which is \underline{not} symmetric.

\medskip
\noindent\textbf{Obs 2.}

\[
\begin{aligned}
0
&=
\frac{\partial}{\partial\theta}
\underbrace{\int f(x,\theta)\,dx}_{=1\ \forall\theta}
=
\int\frac{\partial}{\partial\theta}f(x,\theta)\,dx
\\
&=
\int
\frac{\partial\log f(x,\theta)}{\partial\theta}
f(x,\theta)\,dx
\\
&=
\mathbb{E}_{\theta}
\left[
\frac{\partial\log f(X,\theta)}{\partial\theta}
\right].
\end{aligned}
\]

[There are some conditions for this.
(Leibniz rule). See P \& M.]

``$S_n(\theta)$ is centred when $X\sim P_\theta$.''

\medskip
\noindent\textbf{Obs 3.}

\medskip
\noindent\textbf{Def 5.}
Let $\Theta\subseteq\mathbb{R}^d$.
For any $\theta\in\operatorname{int}(\Theta)$,
the \underline{Fisher Information matrix} is
\[
I(\theta)
=
\mathbb{E}_{\theta}
\left[
\frac{\partial}{\partial\theta}\log f(X,\theta)
\cdot
\left(
\frac{\partial}{\partial\theta}\log f(X,\theta)
\right)^T
\right],
\qquad
X\sim P_\theta.
\]

\medskip
\noindent\textbf{Rmk.}
This is the covariance matrix of
\[
S_n(\theta)
=
\frac{\partial}{\partial\theta}\log f(X,\theta)
\]
when $X\sim P_\theta$.

\medskip
\noindent\textbf{Prop 1.}
For all $\theta\in\operatorname{int}(\Theta)$, in a ``regular'' model,
\[
I(\theta)
=
-\mathbb{E}_{\theta}
\left[
\frac{\partial^2}{\partial\theta\,\partial\theta^T}
\log f(X,\theta)
\right]
\]
\[
\left(
I_{ij}(\theta)
=
-\mathbb{E}_{\theta}
\left[
\frac{\partial^2}{\partial\theta_i\,\partial\theta_j}
\log f(X,\theta)
\right]
\right).
\]

\medskip
\noindent\textbf{Proof.}
\[
0
=
\frac{\partial^2}{\partial\theta\,\partial\theta^T}
\int f(x,\theta)\,dx
=
\int
\frac{\partial^2}{\partial\theta\,\partial\theta^T}
f(x,\theta)\,dx.
\]

[When this equality holds, we say the model is ``regular''.]

Now, we have
\[
\begin{aligned}
\frac{\partial^2}{\partial\theta\,\partial\theta^T}
\log f(x,\theta)
&=
\frac{\partial}{\partial\theta^T}
\left[
\frac{1}{f(x,\theta)}
\frac{\partial}{\partial\theta}f(x,\theta)
\right]
\\
&=
-\frac{1}{f(x,\theta)^2}
\frac{\partial}{\partial\theta}f(x,\theta)
\left(
\frac{\partial}{\partial\theta}f(x,\theta)
\right)^T
\\
&\qquad
+
\frac{1}{f(x,\theta)}
\frac{\partial^2}{\partial\theta\,\partial\theta^T}
f(x,\theta).
\end{aligned}
\]

Take $-\mathbb{E}_{\theta}[\cdot]$ of both sides.
\[
\begin{aligned}
&-\mathbb{E}_{\theta}
\left[
\frac{\partial^2}{\partial\theta\,\partial\theta^T}
\log f(X,\theta)
\right]
\\
&\qquad=
\mathbb{E}_{\theta}
\left[
\frac{1}{f(X,\theta)^2}
\frac{\partial}{\partial\theta}f(X,\theta)
\left(
\frac{\partial}{\partial\theta}f(X,\theta)
\right)^T
\right]
+0
\\
&\qquad=
\mathbb{E}_{\theta}
\left[
\frac{\partial}{\partial\theta}\log f(X,\theta)
\left(
\frac{\partial}{\partial\theta}\log f(X,\theta)
\right)^T
\right]
=
I(\theta).
\end{aligned}
\]
\hfill$\square$

\medskip
\noindent\textbf{Tensorisation.}

Suppose the observation $X$ consists of $n$ random variables
$X_1,\ldots,X_n$. Then, the F.I.\ matrix is
\[
I_n(\theta)
=
\mathbb{E}_{\theta}
\left[
\frac{\partial}{\partial\theta}
\log f(X_1,\ldots,X_n,\theta)
\left(
\frac{\partial}{\partial\theta}
\log f(X_1,\ldots,X_n,\theta)
\right)^T
\right].
\]

If $X_1,\ldots,X_n$ are iid such that
\[
f(X_1,\ldots,X_n,\theta)
=
\prod_{i=1}^{n}f(X_i,\theta),
\]
where $\{f(x,\theta):\theta\in\Theta\}$ is a model for $X_1$,
then the F.I.\ is
\[
\begin{aligned}
I_n(\theta)
&=
\mathbb{E}_{\theta}
\left[
\sum_{i,j}
\underbrace{
\frac{\partial}{\partial\theta}\log f(X_i,\theta)
}_{\substack{\text{mean }0\text{ under}\\P_\theta}}
\left(
\frac{\partial}{\partial\theta}\log f(X_j,\theta)
\right)^T
\right]
\\
&=
\mathbb{E}_{\theta}
\left[
\sum_{i=1}^{n}
\frac{\partial}{\partial\theta}\log f(X_i,\theta)
\left(
\frac{\partial}{\partial\theta}\log f(X_i,\theta)
\right)^T
\right]
\\
&=
nI(\theta),
\end{aligned}
\]
where $I(\theta)$ is the Fisher Information for a single observation.

\subsection*{Recap of Information Geometry}

\paragraph{Observation 1.}
\[
\operatorname{KL}(P_{\theta_0},P_\theta)
=
\mathbb{E}_{\theta_0}
\left[
\log\frac{f(X,\theta_0)}{f(X,\theta)}
\right]
\geq 0,
\]
minimised at $\theta=\theta_0$.

\paragraph{Observation 2.}
\[
\mathbb{E}_{\theta}
\left[
\frac{\partial}{\partial\theta}\log f(X,\theta)
\right]
=0,
\qquad
\forall\theta\in\operatorname{int}(\Theta).
\]

\paragraph{Observation 3.}
\[
I(\theta)
:=
\operatorname{Var}_{\theta}
\left[
\frac{\partial}{\partial\theta}\log f(X,\theta)
\right]
=
-\mathbb{E}_{\theta}
\left[
\frac{\partial^2}{\partial\theta\,\partial\theta^{T}}
\log f(X,\theta)
\right],
\qquad
\forall\theta\in\operatorname{int}(\Theta).
\]

\paragraph{Why geometry?}
The F.I., $I(\theta)$, defines a metric tensor on the manifold
\[
\{f(\cdot,\theta):\theta\in\Theta\}.
\]
This is known as the \textbf{Fisher--Rao metric}.
KL divergence is approximated locally by this metric.

\begin{itemize}
    \item \textbf{Reparametrisation invariance.}
    Take the $\operatorname{Bernoulli}(\theta)$ model, where $\theta$
    is the probability of success. Reparametrise it in terms of
    ``log-odds'':
    \[
    \eta=\log\left(\frac{\theta}{1-\theta}\right)\in\mathbb{R}.
    \]
    The manifold
    \[
    \{f(\cdot,\eta):\eta\in\mathbb{R}\},
    \]
    equipped with the metric tensor $I(\eta)$, preserves geodesics,
    path lengths, etc.

    \item \textbf{Sufficient statistic invariance.}
\end{itemize}

\paragraph{Chentsov's theorems.}
These properties characterise the Fisher--Rao metric.

\subsection*{Proposition 2 (Cram\'er--Rao Lower Bound)}

Let
\[
\{f(\cdot,\theta):\theta\in\Theta\subseteq\mathbb{R}\}
\]
be a ``regular'' statistical model.
Let $\tilde{\theta}(X_1,\ldots,X_n)$ be an unbiased estimator based on
i.i.d.\ samples $X_1,\ldots,X_n$:
\[
\mathbb{E}_{\theta}\tilde{\theta}=\theta,
\qquad
\forall\theta\in\Theta.
\]
Then
\[
\operatorname{Var}_{\theta}(\tilde{\theta})
\geq
\frac{1}{nI(\theta)},
\qquad
\forall\theta\in\operatorname{int}(\Theta).
\]

\paragraph{Proof.}
Recall the Cauchy--Schwarz inequality for random variables $Y,Z$:
\[
\bigl(\mathbb{E}[YZ]\bigr)^2
\leq
\mathbb{E}[Y^2]\mathbb{E}[Z^2].
\]
Let
\[
Z=\tilde{\theta}-\theta,
\qquad
Y=\frac{\partial}{\partial\theta}
\log f(X_1,\ldots,X_n;\theta).
\]
Then
\[
\mathbb{E}[Z^2]
=
\operatorname{Var}_{\theta}(\tilde{\theta}),
\qquad
\text{as $\tilde{\theta}$ is unbiased},
\]
and
\[
\mathbb{E}[Y^2]
=
nI(\theta),
\qquad
\text{as the F.I.\ tensorises}.
\]
Moreover,
\[
\begin{aligned}
\mathbb{E}[YZ]
&=
\mathbb{E}_{\theta}
\left[
\tilde{\theta}
\frac{\partial}{\partial\theta}
\log f(X_1,\ldots,X_n;\theta)
\right]
\\
&\quad
-\theta
\underbrace{
\mathbb{E}_{\theta}
\left[
\frac{\partial}{\partial\theta}
\log f(X_1,\ldots,X_n;\theta)
\right]
}_{=0}
\\
&=
\int_{\mathcal{X}^n}
\tilde{\theta}(x_1,\ldots,x_n)
\frac{\partial f(x_1,\ldots,x_n;\theta)}{\partial\theta}
\frac{f(x_1,\ldots,x_n;\theta)}
     {f(x_1,\ldots,x_n;\theta)}
\,dx_1\cdots dx_n
\\
&=
\frac{\partial}{\partial\theta}
\underbrace{
\int_{\mathcal{X}^n}
\tilde{\theta}\,
f(x_1,\ldots,x_n;\theta)
\,dx_1\cdots dx_n
}_{=\mathbb{E}_{\theta}\tilde{\theta}=\theta}
\qquad
\text{(by regularity)}
\\
&=
\frac{\partial\theta}{\partial\theta}
=1.
\end{aligned}
\]
Plug these expectations into the Cauchy--Schwarz inequality and rearrange.
\hfill$\square$

\paragraph{Remark.}
Sometimes, the MLE is unbiased and attains the Cram\'er--Rao lower
bound, but not always. However, we will show this is true asymptotically.

If $\Theta\subseteq\mathbb{R}^p$ with $p>1$, let
\[
\Phi:\Theta\to\mathbb{R}
\]
be a scalar function of $\theta$.
Consider an unbiased estimator $\tilde{\Phi}(X_1,\ldots,X_n)$.
Then one can show that
\[
\operatorname{Var}_{\theta}(\tilde{\Phi})
\geq
\frac{1}{n}
\left(
\frac{\partial\Phi(\theta)}{\partial\theta}
\right)^{T}
I(\theta)^{-1}
\frac{\partial\Phi(\theta)}{\partial\theta},
\qquad
\forall\theta\in\operatorname{int}(\Theta).
\]

\paragraph{Corollary 1.}
If $\tilde{\theta}$ is not unbiased, we have
\[
\operatorname{Var}_{\theta}(\tilde{\theta})
\geq
\frac{
\displaystyle
\frac{\partial}{\partial\theta}
\mathbb{E}_{\theta}\tilde{\theta}
}{
nI(\theta)
},
\qquad
\forall\theta\in\operatorname{int}(\Theta).
\]

\paragraph{Example.}
Take $\Phi(\theta)=\alpha^{T}\theta$ for some
$\alpha\in\mathbb{R}^p$.
In this case,
\[
\frac{\partial\Phi(\theta)}{\partial\theta}=\alpha,
\]
and
\[
\operatorname{Var}_{\theta}(\tilde{\Phi})
\geq
\frac{1}{n}\alpha^{T}I(\theta)^{-1}\alpha.
\]

\subsection*{Example Sheet 1}

We observe a single pair
\[
(X_1,X_2)
\sim
N\left(
\begin{pmatrix}
\theta_1\\
\theta_2
\end{pmatrix},
\Sigma
\right),
\]
where $\Sigma\succ0$ is known.
We wish to estimate $\theta_1$.

\paragraph{Case 1.}
Assume that $\theta_2$ is known.
In this case, the model has one parameter and F.I.\ matrix
\[
I_1(\theta_1).
\]

\paragraph{Case 2.}
Assume that $\theta_2$ is unknown.
For any estimator
\[
\tilde{\theta}
=
\begin{pmatrix}
\tilde{\theta}_1\\
\tilde{\theta}_2
\end{pmatrix},
\]
we have
\[
\tilde{\theta}_1
=
\begin{pmatrix}
1\\
0
\end{pmatrix}^{T}
\tilde{\theta}.
\]
So, if $\tilde{\theta}$ is unbiased, we have the Cram\'er--Rao lower bound
\[
\operatorname{Var}_{\theta}(\tilde{\theta}_1)
\geq
\begin{pmatrix}
1\\
0
\end{pmatrix}^{T}
I(\theta)^{-1}
\begin{pmatrix}
1\\
0
\end{pmatrix}
=
\bigl[I(\theta)^{-1}\bigr]_{11}.
\]

\paragraph{Goal.}
Show
\[
I_1(\theta_1)^{-1}
<
\bigl[I(\theta)^{-1}\bigr]_{11},
\]
unless $\Sigma$ is diagonal ($X_1,X_2$ independent).

\section*{II. Asymptotic Theory of the MLE}

All reasonable estimators are asymptotically unbiased:
\[
\mathbb{E}_{\theta}\tilde{\theta}
\xrightarrow{n\to\infty}
\theta,
\qquad
\forall\theta\in\Theta.
\]

We wish to prove that estimators are \textbf{consistent}:
\[
\tilde{\theta}\xrightarrow{?}\theta
\qquad
\text{as }n\to\infty
\quad
\text{under }P_{\theta},
\qquad
\forall\theta\in\Theta.
\]

For consistent estimators, a reasonable notion of optimality is
\textbf{asymptotic efficiency}:
\[
n\operatorname{Var}_{\theta}(\tilde{\theta})
\xrightarrow{n\to\infty}
I(\theta)^{-1},
\qquad
\forall\theta\in\operatorname{int}(\Theta).
\]

We will show that the MLE is asymptotically efficient and, furthermore,
\[
\hat{\theta}_{\mathrm{MLE}}
\approx
N\left(\theta,n^{-1}I(\theta)^{-1}\right),
\qquad
\forall\theta.
\]
This distributional convergence can be used for statistical inference.

\subsection*{Stochastic Convergence}

\paragraph{Definition 1.}
Let $(X_n:n\in\mathbb{N})$ be a collection of random variables in some
probability space $(\Omega,\mathcal{F},\mathbb{P})$.

\begin{itemize}
    \item We say that $X_n$ converges almost surely to a random variable
    $X$ on $(\Omega,\mathcal{F},\mathbb{P})$, or
    \[
    X_n\xrightarrow{\mathrm{a.s.}}X,
    \]
    if and only if
    \[
    \mathbb{P}\left(
    \left\{
    \omega:
    \left\|X_n(\omega)-X(\omega)\right\|_2
    \xrightarrow{n\to\infty}0
    \right\}
    \right)
    =1.
    \]

    \item We say that $X_n$ converges to $X$ in probability, or
    \[
    X_n\xrightarrow{P}X,
    \]
    if, for every $\varepsilon>0$,
    \[
    \mathbb{P}\left(
    \|X_n-X\|_2>\varepsilon
    \right)
    \xrightarrow{n\to\infty}0.
    \]
\end{itemize}

Recall: If $(X_n),X$ are random vectors in $\mathbb{R}^k$, we say
\begin{itemize}
    \item
    $X_n \xrightarrow{\mathrm{a.s.}} X$ iff
    \[
    \mathbb{P}\left(
    \|X_n-X\|_2 \xrightarrow{n\to\infty} 0
    \right)=1.
    \]

    \item
    $X_n \xrightarrow{P} X$ iff
    \[
    \forall \varepsilon>0,\qquad
    \mathbb{P}\left(\|X_n-X\|_2>\varepsilon\right)
    \xrightarrow{n\to\infty}0.
    \]
\end{itemize}

\noindent
\underline{Rmk.}
\[
X_n \xrightarrow{\mathrm{a.s.}/P} X
\quad\text{iff}\quad
X_{n,i} \xrightarrow{\mathrm{a.s.}/P} X_i
\qquad \forall\,1\leq i\leq k.
\]

\[
\left[
X\leq t
\iff
X_i\leq t_i
\quad \forall\,1\leq i\leq k
\right]
\]

\noindent
\underline{Def 2.}
We say that $X_n$ converges \underline{in distribution} to $X$,
or $X_n\xrightarrow{d}X$, if for all $t\in\mathbb{R}^k$
where the function
\[
t\mapsto\mathbb{P}(X\leq t)
\]
is continuous, we have
\[
\mathbb{P}(X_n\leq t)
\xrightarrow{n\to\infty}
\mathbb{P}(X\leq t).
\]

Some facts from P \& M.

\medskip
\noindent
\underline{Prop 1.}

\begin{enumerate}
    \item[(a)]
    \[
    X_n\xrightarrow{\mathrm{a.s.}}X
    \implies
    X_n\xrightarrow{P}X
    \implies
    X_n\xrightarrow{d}X.
    \]

    \item[(b)]
    \underline{Continuous Mapping Theorem} (CMT).
    Let $(X_n),X$ take values in
    $\mathcal{X}\subseteq\mathbb{R}^k$,
    let $g:\mathcal{X}\to\mathbb{R}$ be continuous.
    Then,
    \[
    X_n\xrightarrow{\mathrm{a.s.}/P/d}X
    \implies
    g(X_n)\xrightarrow{\mathrm{a.s.}/P/d}g(X).
    \]

    \item[(c)]
    \underline{Slutsky's Lemma.}
    Suppose that
    \[
    X_n\xrightarrow{d}X,
    \qquad
    Y_n\xrightarrow{d}c,
    \]
    where $c$ is constant, then
    \begin{enumerate}
        \item[(i)]
        \[
        Y_n\xrightarrow{P}c.
        \]

        \item[(ii)]
        \[
        X_n+Y_n\xrightarrow{d}X+c.
        \]

        \item[(iii)]
        If $k=1$,
        \[
        X_nY_n\xrightarrow{d}X\cdot c,
        \]
        \[
        \frac{X_n}{Y_n}\xrightarrow{d}\frac{X}{c}
        \qquad\text{if }c\neq0.
        \]

        \item[(iv)]
        Let $(A_n)$ be a sequence of random matrices in
        $\mathbb{R}^{m\times k}$, and suppose,
        \[
        (A_n)_{ij}\xrightarrow{P}A_{ij}
        \qquad \forall i,j
        \]
        where $A$ is deterministic. Then,
        \[
        A_nX_n\xrightarrow{d}AX.
        \]
    \end{enumerate}

    \item[(d)]
    If $X_n\xrightarrow{d}X$, then $(X_n)$ is
    ``bounded in probability'' or $X_n=O_P(1)$, i.e.\
    for all $\varepsilon>0$,
    $\exists\,M_\varepsilon>0$, $N_\varepsilon\in\mathbb{N}$ s.t.
    \[
    \mathbb{P}\left(\|X_n\|_2>M_\varepsilon\right)
    <\varepsilon
    \qquad \forall n\geq N_\varepsilon.
    \]
\end{enumerate}

\noindent
\underline{2. LLN \& CLT}

\medskip
\noindent
Let $X_1,X_2,\ldots$ be iid copies of a r.v.\ $X$ in
$\mathbb{R}^k$.
When $k=1$, the Weak Law of large numbers says that if
$\operatorname{Var}(X)<\infty$, then
\[
\bar{X}_n:=\frac{1}{n}\sum_{i=1}^n X_i
\xrightarrow{P}\mathbb{E}X.
\]
Indeed by Chebyshev's inequality,
\[
\forall\varepsilon>0,\qquad
\mathbb{P}\left(
|\bar{X}_n-\mathbb{E}X|>\varepsilon
\right)
\leq
\frac{\operatorname{Var}(\bar{X}_n)}{\varepsilon^2}
=
\frac{\operatorname{Var}(X)}{n\varepsilon^2}
\xrightarrow{n\to\infty}0.
\]

\medskip
\noindent
\underline{Theorem 1} (Strong law of large numbers).
If $\mathbb{E}\left(\|X_n\|_2\right)<\infty$, then
\[
\bar{X}_n\xrightarrow{\mathrm{a.s.}}\mathbb{E}[X]
\qquad\text{as }n\to\infty.
\]

\medskip
\noindent
\underline{Theorem 2} (Central Limit Theorem).
When $k=1$, assume in addition that
$\operatorname{Var}(X)=\sigma^2<\infty$, then,
\[
\sqrt{n}\left(\bar{X}_n-\mathbb{E}X\right)
\xrightarrow{d}N(0,\sigma^2).
\]

\noindent
To state a multivariate version, recall that
$Z\sim N(\mu,\Sigma)$ iff
\[
\forall\alpha\in\mathbb{R}^k,\qquad
\alpha^T Z\sim N(\alpha^T\mu,\alpha^T\Sigma\alpha).
\]
If $\Sigma$ is positive definite, then $Z$ has a p.d.f.
\[
f(z)
=
\frac{1}{(2\pi)^{n/2}}
\frac{1}{|\det\Sigma|^{1/2}}
\exp\left\{
-\frac{1}{2}(z-\mu)^T\Sigma^{-1}(z-\mu)
\right\},
\qquad z\in\mathbb{R}^k.
\]

\medskip
\noindent
\underline{Theorem 2} $(k\geq1)$.
If $\operatorname{Var}(X)=\Sigma$ is finite \& pos.\ def.\ then
\[
\sqrt{n}\left(\bar{X}_n-\mathbb{E}X\right)
\xrightarrow{d}N(0,\Sigma).
\]

\medskip
\noindent
\underline{Cor.}
By Prop 1(d),
\[
\sqrt{n}\left(\bar{X}_n-\mathbb{E}X\right)=O_P(1)
\iff
\left(\bar{X}_n-\mathbb{E}X\right)
=
O_P\left(\frac{1}{\sqrt{n}}\right).
\]

\medskip
\noindent
The CLT is useful to define confidence intervals.
Let $X_1,X_2,\ldots$ be iid copies of $X$ with mean $\mu_0$
and $\operatorname{Var}(X)=\sigma^2<\infty$ known.
Define
\[
C_n
=
\left\{
\mu\in\mathbb{R}:
\sqrt{n}\frac{|\bar{X}_n-\mu|}{\sigma}
\leq z_\alpha
\right\}
\]
where $z_\alpha$ is such that
\[
\mathbb{P}(|Z|\leq z_\alpha)=1-\alpha
\qquad\text{if }Z\sim N(0,1).
\]

\[
\begin{aligned}
\mathbb{P}(\mu_0\in C_n)
&=
\mathbb{P}\left(
\sqrt{n}|\bar{X}_n-\mu_0|\leq\sigma z_\alpha
\right)\\
&\xrightarrow{n\to\infty}
\mathbb{P}(|Z|\leq z_\alpha)
\qquad\text{by the CLT}\\
&=1-\alpha.
\end{aligned}
\]

\noindent
Hence, we say that $C_n$ is an ``asymptotic''
$(1-\alpha)$-confidence interval.

\medskip
\noindent
If $\sigma^2$ is unknown, we can define an estimator
\[
\begin{aligned}
\hat{\sigma}^2
&=
\frac{1}{n-1}
\sum_{i=1}^n
\left(
X_i-\underbrace{\bar{X}_n}_{\frac{1}{n}\sum_j X_j}
\right)^2\\
&=
\frac{1}{n-1}
\left(
\sum_{i=1}^n X_i^2-n\bar{X}_n^2
\right)\\
&\xrightarrow{d}
\mathbb{E}(X^2)-(\mathbb{E}X)^2
=
\sigma^2
\end{aligned}
\]
by several applications of the SLLN, the CMT and Slutsky's lemma.

\medskip
\noindent
Define a new random interval
\[
\widetilde{C}_n
=
\left\{
\mu\in\mathbb{R}:
\sqrt{n}\frac{|\bar{X}_n-\mu|}{\hat{\sigma}}
\leq z_\alpha
\right\}.
\]

\noindent
By the CMT, $\hat{\sigma}\xrightarrow{d}\sigma$ constant.
By Slutsky's Lemma,
\[
\sqrt{n}\frac{(\bar{X}_n-\mu_0)}{\hat{\sigma}}
\xrightarrow{d}N(0,1).
\]

\noindent
Hence, $C_n$ is an asymptotic $(1-\alpha)$-confidence interval
for $\mu_0$.

Let $\mathcal{H}$ be a class of functions

We know from the SLLN that

\]

\subsection*{Uniform Laws of Large Numbers}

Let $X$ be a r.v. on $\mathcal{X}$. $(X_i)$ iid copies of $X$.
Let $\mathcal{H}$ be a class of functions $\mathcal{X}\to\mathbb{R}$, s.t.
\[
\mathbb{E}|h(X)|<\infty\qquad\forall h\in\mathcal{H}.
\]
\[
\underline{\mathrm{ULLN}}\qquad
\sup_{h\in\mathcal{H}}\left|
\underbrace{\frac{1}{n}\sum_{i=1}^n h(X_i)}_{=P_nh}
-\underbrace{\mathbb{E}h(X)}_{=Ph}\right|
\xrightarrow{\mathrm{a.s.}}0.
\]
If $\mathcal{H}=\{h_1,\ldots,h_M\}$ is finite, then by the SLLN,
for any $1\leq j\leq M$,
\[
|P_nh_j-Ph_j|\xrightarrow{\mathrm{a.s.}}0
\]
i.e. on an event $A_j$ with $\mathbb{P}(A_j)=1$, $\forall\varepsilon>0$,
$\exists N_j(\varepsilon)$ s.t.
\[
|P_nh_j-Ph_j|<\varepsilon\qquad\forall n>N_j(\varepsilon).
\]
Hence on $A:=\bigcap_{j=1}^M A_j$, we have $\forall\varepsilon>0$,
\[
\max_{1\leq j\leq M}|P_nh_j-Ph_j|<\varepsilon
\qquad\forall n>\max_j N_j(\varepsilon).
\]
As $\mathbb{P}(A)=1$, this means
\[
\sup_{h\in\mathcal{H}}|P_nh-Ph|\xrightarrow{\mathrm{a.s.}}0.
\]
When $|\mathcal{H}|$ is infinite, we will use a ``bracketing argument''.
A \underline{bracket} is a collection of functions
$\{\underline{h}_j,\overline{h}_j\}_{j=1}^M$ such that
\[
\forall j,\qquad\underline{h}_j(x)\leq\overline{h}_j(x)
\qquad\forall x\in\mathcal{X}.
\]
A finite bracket \underline{covers} $\mathcal{H}$ if for all $h\in\mathcal{H}$,
$\exists j\in\{1,\ldots,M\}$ s.t.
\[
\underline{h}_j(x)\leq h(x)\leq\overline{h}_j(x)
\qquad\forall x\in\mathcal{X}.
\]

\paragraph{Prop 2.}
Suppose that $\forall\varepsilon>0$, there is a finite bracket
$\{\underline{h}_j,\overline{h}_j\}_{j=1}^M$ covering $\mathcal{H}$, s.t.
\begin{align*}
\text{(1)}\quad&\mathbb{E}|\underline{h}_j(X)|<\infty,\quad
\mathbb{E}|\overline{h}_j(X)|<\infty\qquad\forall 1\leq j\leq M,\\
\text{(2)}\quad&\mathbb{E}|\overline{h}_j(X)-\underline{h}_j(X)|<\varepsilon.
\end{align*}
Then,
\[
\sup_{h\in\mathcal{H}}|P_nh-Ph|\xrightarrow{\mathrm{a.s.}}0.
\]
\noindent\underline{Proof.}
Take any $\varepsilon>0$, then for any $h\in\mathcal{H}$,
$\exists 1\leq j\leq M$ s.t. $\underline{h}_j\leq h\leq\overline{h}_j$.
Then
\[
\begin{aligned}
P_nh-Ph&\leq P_n\overline{h}_j-P\underline{h}_j\\
&\leq |P_n\overline{h}_j-P\overline{h}_j|
+\underbrace{|P(\overline{h}_j-\underline{h}_j)|}_{\leq\varepsilon}.
\end{aligned}
\]
Similarly,
\[
Ph-P_nh\leq P\overline{h}_j-P_n\underline{h}_j
\leq |P_n\underline{h}_j-P\underline{h}_j|
+\underbrace{|P(\overline{h}_j-\underline{h}_j)|}_{\leq\varepsilon}.
\]
\[
\therefore\quad\sup_{h\in\mathcal{H}}|P_nh-Ph|
\leq\underbrace{\max_{g\in\mathrm{bracket}}|P_ng-Pg|}
_{\xrightarrow{\mathrm{a.s.}}0\text{ by the SLLN as bracket is finite}}
+\varepsilon.
\]
Hence, on an event $A_\varepsilon$ with $\mathbb{P}(A_\varepsilon)=1$, we have
\[
\limsup_{n\to\infty}\sup_{h\in\mathcal{H}}|P_nh-Ph|\leq\varepsilon.
\]
Take sequence $\varepsilon_k\downarrow0$. Let
$A=\bigcap_{k=1}^\infty A_{\varepsilon_k}$, with $\mathbb{P}(A)=1$.
On $A$,
\[
\sup_{h\in\mathcal{H}}|P_nh-Ph|\longrightarrow0.\qquad\square
\]

\subsection*{3 Consistency of the MLE}

\paragraph{Def 3.}
We say an estimator $\hat{\theta}(X_1,\ldots,X_n)$ is \underline{consistent}
if $\hat{\theta}\xrightarrow{P}\theta$ under $P_\theta$ $\forall\theta\in\Theta$.

\medskip\noindent\underline{Assumptions.}
Let $\{f(\cdot,\theta):\theta\in\Theta\}$ be a model of pdfs or pmfs on
$\mathcal{X}\subseteq\mathbb{R}^d$. We'll assume
\begin{enumerate}
\item[(1)] $f(x,\theta)>0\qquad\forall x\in\mathcal{X},\ \theta\in\Theta$.
\item[(2)] $\theta\mapsto f(x,\theta)$ is continuous $\forall x\in\mathcal{X}$.
\item[(3)] $\Theta$ is compact.
\item[(4)] \underline{Strict identifiability}: $f(\cdot,\theta)=f(\cdot,\theta')
\implies\theta=\theta'$.
\item[(5)] $\displaystyle\mathbb{E}_\theta
\underbrace{\sup_{\theta'\in\Theta}|\log f(X,\theta')|}_{M(X)}<\infty
\qquad\forall\theta\in\Theta$.
\end{enumerate}

Recall that (5) allows us to define
$\ell(\theta):=\mathbb{E}_{\theta_0}\log f(X,\theta)$ for a true parameter $\theta_0$.
Furthermore, by (4) $\ell(\theta)$ is maximised uniquely at $\theta=\theta_0$.

By (2) \& (5), we have that $\ell$ is continuous.
Indeed take $(\theta_i)$, $\theta_i\to\theta$, then
\[
\begin{aligned}
|\ell(\theta_i)-\ell(\theta)|
&=\left|\mathbb{E}_{\theta_0}\left[
\underbrace{\log f(X,\theta_i)-\log f(X,\theta)}
_{\xrightarrow{\mathrm{a.s.}}0\text{ by (2)}}\right]\right|\\
&\longrightarrow0\quad\begin{gathered}
\text{by the Dominated Convergence Theorem (P\&M)}\\
\text{using the domination condition (5).}
\end{gathered}
\end{aligned}
\]
In maximum likelihood we maximise
\[
\overline{\ell}_n(\theta)=\frac{1}{n}\sum_{i=1}^n\log f(X_i,\theta),
\qquad\text{a ``sample'' version of }\ell(\theta).
\]

\paragraph{Lemma.}
If $(X_i)$ are iid with law $P_{\theta_0}$, we have the ULLN
\[
\sup_{\theta\in\Theta}\left|
\underbrace{\overline{\ell}_n(\theta)}_{=P_nh_\theta}
-\underbrace{\ell(\theta)}_{=Ph_\theta}\right|
\xrightarrow{\mathrm{a.s.}}0
\]
where $h_\theta:x\mapsto\log f(x,\theta)$.
Here $\mathcal{H}=\{h_\theta:\theta\in\Theta\}$.

\medskip\noindent\underline{*Proof sketch*}
To apply Prop 2, we need to construct for any $\varepsilon>0$ a finite bracket
satisfying the hypotheses (1) \& (2).
Take any $\theta\in\Theta$, let
$B_{\theta,\delta}=\{\theta':\|\theta-\theta'\|\leq\delta\}$, let
\[
\overline{h}_\theta^{(\delta)}=\sup_{\theta'\in B_{\theta,\delta}}h_{\theta'}(x),
\qquad
\underline{h}_\theta(x)=\inf_{\theta'\in B_{\delta,\theta}}h_{\theta'}(x).
\]
\begin{center}
\includegraphics[width=0.30\textwidth]{figures/principles-of-statistics/lecture5-compactness.png}
\end{center}
By def
\[
\underline{h}_\theta\leq h_{\theta'}\leq\overline{h}_\theta
\qquad\forall\theta'\in B_{\delta,\theta}.
\]
Hypothesis (1) is given by Assumption (5).
We'll choose $\delta$ small enough so that we can obtain Hypothesis 2:
\[
\mathbb{E}|\overline{h}_\theta(X)-\underline{h}_\theta(X)|<\varepsilon
\]
(To do this carefully, apply the DCT using assumptions (2) \& (5).)

For each $\theta\in\Theta$ we obtain a pair $(\underline{h}_\theta,\overline{h}_\theta)$
sandwiching functions in a ``ball'' around $h_\theta$.
By compactness, $\exists$ a finite bracket
$\{\overline{h}_{\theta_j},\underline{h}_{\theta_j}\}_{j=1}^M$ s.t.
$\forall\theta\in\Theta$, $\exists j$ s.t.
\[
\underline{h}_{\theta_j}\leq h_\theta\leq\overline{h}_{\theta_j}.
\]
This allows us to apply prop 2.\quad$\square$

\paragraph{Recall.}
We stated 5 assumptions for consistency of the MLE.

\noindent\underline{Implications}
\begin{itemize}
\item $\ell(\theta)=\mathbb{E}_{\theta_0}[\log f(X,\theta)]<\infty$
$\forall\theta$, maximised uniquely at $\theta_0$.
\item $\ell$ is continuous on $\Theta$.
\item \underline{ULLN}
\[
\sup_{\theta\in\Theta}|\overline{\ell}_n(\theta)-\ell(\theta)|
\xrightarrow{\mathrm{a.s.}/P/d}0.
\]
\end{itemize}

\paragraph{Theorem 3.}
Suppose that $\{f(\cdot,\theta):\theta\in\Theta\}$ satisfies the stated
assumptions and $X_1,X_2,\ldots$ are iid $P_{\theta_0}$.
Then an MLE $\hat{\theta}_n$ based on $X_1,\ldots,X_n$ exists and
\[
\hat{\theta}_n\xrightarrow{P}\theta_0.
\]

\paragraph{Proof.}
\[
\overline{\ell}_n(\theta)=\frac{1}{n}\sum_{i=1}^n\log f(X_i,\theta)
\]
is continuous, so it attains a supremum in $\Theta$ compact at some
$\hat{\theta}_n\in\Theta$.

Define
\[
\Theta_\varepsilon=\{\theta\in\Theta:\|\theta-\theta_0\|_2\geq\varepsilon\}
\]
and note $\Theta_\varepsilon$ is compact (we remove an open ball from
$\Theta$ compact). We'll show that
\[
\mathbb{P}(\hat{\theta}_n\in\Theta_\varepsilon)\longrightarrow0
\qquad\text{as }n\to\infty.
\]
If this holds for arbitrary $\varepsilon>0$,
$\hat{\theta}_n\xrightarrow{P}\theta_0$.

\includegraphics[width=0.40\textwidth]{figures/principles-of-statistics/lecture6-consistency.png}
\end{center}

\[
\sup_{\theta\in\Theta_\varepsilon}\ell(\theta)=:c(\varepsilon)<\ell(\theta_0).
\]
Choose
\[
\delta(\varepsilon)<\frac{\ell(\theta_0)-c(\varepsilon)}{2}.
\]
Define an event
\[
A_\varepsilon=\left\{\sup_{\theta\in\Theta}
|\overline{\ell}_n(\theta)-\ell(\theta)|\leq\delta(\varepsilon)\right\}.
\]
By the ULLN, $\mathbb{P}(A_\varepsilon)\to1$ as $n\to\infty$.
\[
\begin{aligned}
\sup_{\theta\in\Theta_\varepsilon}\overline{\ell}_n(\theta)
&\leq\sup_{\theta\in\Theta_\varepsilon}\ell(\theta)
+\underbrace{\sup_{\theta\in\Theta_\varepsilon}
(\overline{\ell}_n(\theta)-\ell(\theta))}_{\leq\delta(\varepsilon)\text{ on }A_\varepsilon}\\
&\leq c(\varepsilon)+\delta(\varepsilon)\\
&<\ell(\theta_0)-\delta(\varepsilon)
\qquad\text{(by def of $\delta(\varepsilon)$)}\\
&\leq\overline{\ell}_n(\theta_0).
\end{aligned}
\]
Hence, on $A_\varepsilon$, $\hat{\theta}_n\notin\Theta_\varepsilon$.
In other words $A_\varepsilon\subseteq\{\hat{\theta}_n\notin\Theta_\varepsilon\}$.
Therefore
\[
\mathbb{P}(\hat{\theta}_n\notin\Theta_\varepsilon)
\geq\mathbb{P}(A_\varepsilon)\longrightarrow1\qquad\text{as }n\to\infty.
\]
Finally
\[
\mathbb{P}(\hat{\theta}_n\in\Theta_\varepsilon)\longrightarrow0
\qquad\text{as }n\to\infty.\qquad\square
\]

\paragraph{Rmk.}
If we assume that $\theta\mapsto\log f(x,\theta)$ is differentiable for
all $x\in\mathcal{X}$, and $\frac{\partial\overline{\ell}_n(\theta)}{\partial\theta}$
satisfies a ULLN and its limit $\frac{\partial\ell(\theta)}{\partial\theta}$
has a single $0$ at $\theta_0$, there is a simpler argument to show consistency.
Can deal with noncompact $\Theta$ in some simple models.

\subsection*{4. Asymptotic Normality of the MLE}
Fluctuations of $\sqrt{n}(\hat{\theta}_n-\theta_0)$.

\paragraph{Theorem 4.}
Suppose that $\{f(\cdot,\theta):\theta\in\Theta\}$ satisfies the 9 stated
assumptions. Let $\hat{\theta}_n$ be an MLE based on $X_1,\ldots,X_n$
iid $P_{\theta_0}$. Then as $n\to\infty$,
\[
\sqrt{n}(\hat{\theta}_n-\theta_0)\xrightarrow{d}
N(0,I^{-1}(\theta_0)).
\]

The nine assumptions are the five assumptions for consistency stated above,
together with the following four additional regularity assumptions (R2).

\medskip\noindent\textbf{Regularity assumptions (R2).} Let the statistical model be such that:
\begin{enumerate}
\item[(1)] $\theta_0\in\operatorname{int}\Theta$;
\item[(2)] there exists an open neighbourhood $N$ of $\theta_0$ on which
$\theta\mapsto f(x,\theta)$ is twice continuously differentiable for all $x\in\mathcal{X}$;
\item[(3)] $I(\theta_0)$ exists and is invertible;
\item[(4)]
\[
\int_{\mathcal{X}}\sup_{\theta\in\Theta}
\|\nabla_\theta\log f(x,\theta)\|_2\,dx<\infty
\]
\[
\mathbb{E}_{\theta_0}\left(\sup_{\theta\in\Theta}
\|\nabla_\theta^2\log f(X,\theta)\|_2\right)<\infty
\]
\[
\int_{\mathcal{X}}\sup_{\theta\in\Theta}
\|\nabla_\theta^2\log f(x,\theta)\|_2\,dx<\infty.
\]
\end{enumerate}
(In fact the above can be weakened by replacing $\Theta$ above with any compact set $K$ with
non-empty interior that contains $\theta_0$.)

\noindent\underline{Theorem 4 (Asymp. Normality of MLE).}\quad
Suppose $X_1,\ldots,X_n\overset{\mathrm{iid}}{\sim}P_{\theta_0}$, and the 9 stated
assumptions hold. Then the MLE $\hat{\theta}_n$ satisfies
\[
\sqrt{n}(\hat{\theta}_n-\theta_0)\xrightarrow{d}
N(0,I(\theta_0)^{-1}).
\]

\paragraph{Proof.}
Recall that
\[
\ell_n(\theta)=\frac1n\sum_{i=1}^n\log f(X_i,\theta).
\]

\paragraph{Step 1. Taylor expansion.}
Choose a compact neighbourhood $K\subset\operatorname{int}\Theta$
of $\theta_0$ on which the regularity conditions hold,
and let
\[
E_n=\{\|\hat{\theta}_n-\theta_0\|<\varepsilon\},
\]
where $\varepsilon>0$ is sufficiently small that the
line segment $[\theta_0,\hat{\theta}_n]$ lies in $K$ on $E_n$.
By consistency of MLE,
\[
\mathbb{P}(E_n)\to1.
\]
On $E_n$, $\|\hat{\theta}_n-\theta_0\|<\varepsilon$, so
$B_\varepsilon(\theta_0)\subseteq\operatorname{int}(K)\subset\operatorname{int}(\Theta)$
given $\varepsilon$ small.
Hence, the MLE is the interior maximiser.
So
\[
\nabla\ell_n(\hat{\theta}_n)=0.
\]
Apply MVT componentwise:
\[
\nabla\ell_n(\hat{\theta}_n)=0
=\nabla\ell_n(\theta_0)+A_n(\hat{\theta}_n-\theta_0),
\]
where
\[
(A_n)_{jk}=\frac1n\sum_{i=1}^n
\frac{\partial^2}{\partial\theta_j\partial\theta_k}
\log\bigl(f(X_i,\theta^{(j)})\bigr),
\qquad
\theta^{(j)}\in[\theta_0,\hat{\theta}_n].
\]

\paragraph{Step 2. Prove $A_n\xrightarrow{P}-I(\theta_0)$.}
Fix $j,k$, define
\[
g(x,\theta)=\frac{\partial^2}{\partial\theta_j\partial\theta_k}
\log f(x,\theta),
\qquad
\bar{g}(\theta)=\mathbb{E}_{\theta_0}[g(X,\theta)].
\]
By equivalent form of Fisher information,
\[
\bar{g}(\theta_0)=-I(\theta_0)_{jk}.
\]
Hence,
\[
\begin{aligned}
|(A_n)_{jk}+I(\theta_0)_{jk}|
&=\left|\frac1n\sum_{i=1}^n g(X_i,\theta^{(j)})-\bar{g}(\theta_0)\right|\\
&\leq\underbrace{\sup_{\theta\in K}
\left|\frac1n\sum_{i=1}^n g(X_i,\theta)-\bar{g}(\theta)\right|}_{\text{(I)}}
+\underbrace{|\bar{g}(\theta^{(j)})-\bar{g}(\theta_0)|}_{\text{(II)}}.
\end{aligned}
\]
For (I), by continuity of $g(x,\cdot)$, compactness of $K$, and
\[
\mathbb{E}_{\theta_0}\left[\sup_{\theta\in K}|g(X,\theta)|\right]<\infty,
\]
and the ``bracket'' version of ULLN, we can in a similar way deduce that
$g$ has ULLN here, namely
\[
\text{(I)}\xrightarrow{\mathrm{a.s.}}0.
\]
For (II), since $\theta^{(j)}\in[\theta_0,\hat{\theta}_n]$,
\[
\|\theta^{(j)}-\theta_0\|
\leq\|\hat{\theta}_n-\theta_0\|\xrightarrow{P}0
\]
by consistency of MLE.
Moreover, DCT implies that $\bar{g}$ is cts.\ at $\theta_0$,
thus by CMT,
\[
\text{(II)}\xrightarrow{P}0.
\]
Thus, for $\forall j,k$,
\[
(A_n)_{jk}\xrightarrow{P}-I(\theta_0)_{jk}.
\]
Result follows.\hfill$\square$

Since $I(\theta_0)$ is invertible, by CMT on $\det(\cdot)$,
$A_n$ is invertible with probability tending to 1
(i.e., in probability).

\paragraph{Step 3. Apply multivariate CLT.}
Define
\[
S_i=\nabla_\theta\log f(X_i,\theta_0).
\]
By regularity conditions,
\[
\mathbb{E}_{\theta_0}[S_i]=0,
\qquad
\operatorname{Var}_{\theta_0}(S_i)=I(\theta_0).
\]
By CLT,
\[
\sqrt{n}\,\nabla\ell_n(\theta_0)
=\frac1{\sqrt{n}}\sum_{i=1}^n S_i
\xrightarrow{d}N(0,I(\theta_0)).
\]

\paragraph{Step 4. Apply Slutsky's Lemma.}
On $E_n$, whenever $A_n$ is invertible,
Taylor expansion gives
\[
\sqrt{n}(\hat{\theta}_n-\theta_0)
=-A_n^{-1}\sqrt{n}\,\nabla\ell_n(\theta_0). \tag{$*$}
\]
By Slutsky's Lemma, and claim,
\[
-A_n^{-1}\sqrt{n}\,\nabla\ell_n(\theta_0)
\xrightarrow{d}N(0,I(\theta_0)^{-1}I(\theta_0)I(\theta_0)^{-1}).
\]
(Using fact:
$X_n\sim N(\mu,\Sigma)\Rightarrow AX_n\sim N(A\mu,A\Sigma A^T)$.)
So
\[
-A_n^{-1}\sqrt{n}\,\nabla\ell_n(\theta_0)
\xrightarrow{d}N(0,I(\theta_0)^{-1}).
\]
Finally, since $\mathbb{P}(E_n)\to1$ and
$\mathbb{P}(A_n\text{ invertible})\to1$,
$(*)$ holds with probability tending to 1.
So the difference $\xrightarrow{P}0$.
So using Slutsky again,
\[
X_n\xrightarrow{d}X,\quad Y_n\xrightarrow{P}0
\quad\Rightarrow\quad X_n+Y_n\xrightarrow{d}X.
\]
Result follows.\hfill$\square$

In models with the ``local asymptotic normality'', any estimator

\noindent\underline{Ex}\quad Hodge's estimator

Let $\hat{\theta}_n$ be the MLE in a regular model s.t.
\[
\sqrt{n}(\hat{\theta}_n-\theta_0)\xrightarrow{d}N(0,I^{-1}(\theta_0))
\qquad\forall\theta_0\in\operatorname{int}(\Theta).
\]
Define
\[
\tilde{\theta}_n=
\begin{cases}
0&\text{if }|\hat{\theta}_n|<n^{-1/4},\\
\hat{\theta}_n&\text{o.w.}
\end{cases}
\]
If $\theta_0\neq0$, under $P_{\theta_0}$,
\[
\begin{aligned}
\mathbb{P}(\tilde{\theta}_n\neq\hat{\theta}_n)
&=\mathbb{P}\bigl(\underbrace{|\hat{\theta}_n|}
_{|\theta_0|-|\hat{\theta}_n-\theta_0|\leq}<n^{-1/4}\bigr)\\
&\leq\mathbb{P}\bigl(|\theta_0|-|\hat{\theta}_n-\theta_0|<n^{-1/4}\bigr)\\
&=\mathbb{P}\bigl(|\hat{\theta}_n-\theta_0|>|\theta_0|-n^{-1/4}\bigr)
\xrightarrow{n\to\infty}0
\qquad\text{as }\hat{\theta}_n\text{ consistent}.
\end{aligned}
\]
Therefore, under $P_{\theta_0}$,
$\sqrt{n}(\tilde{\theta}_n-\theta_0)\xrightarrow{d}N(0,I^{-1}(\theta_0))$.
However, if $\theta_0=0$,
\[
\begin{aligned}
\mathbb{P}(\hat{\theta}_n=\tilde{\theta}_n)
&=\mathbb{P}(|\hat{\theta}_n|\geq n^{-1/4})\\
&=\mathbb{P}\bigl(\sqrt{n}\,|\hat{\theta}_n-\theta_0|\geq n^{1/4}\bigr).
\end{aligned}
\]
Recall that by Prop 1 in lecture 4,
$\sqrt{n}(\hat{\theta}_n-\theta_0)=O_{\mathbb{P}}(1)$, which implies
the RHS $\xrightarrow{n\to\infty}0$.
In other words,
\[
\mathbb{P}(\tilde{\theta}_n=0)\xrightarrow{n\to\infty}1
\quad\text{or}\quad
\sqrt{n}(\tilde{\theta}_n-\theta_0)\xrightarrow{d}0.
\]
Hence the asymptotic variance is $0<I^{-1}(\theta_0)$ in general.

\subsection*{5. Asymptotic Inference}

\noindent\underline{Thm 4:}
\[
\sqrt n(\hat\theta_n-\theta_0)\xrightarrow{d}N(0,I^{-1}(\theta_0)).
\]
To construct a confidence set for $\theta_0$, we could use a pivot:
\[
I^{1/2}(\theta_0)\sqrt n(\hat\theta_n-\theta_0)
\xrightarrow{d}N(0,I)\qquad\text{by CMT}.
\]
\noindent Problem: we don't know $\theta_0$, so can't compute $I(\theta_0)$.

Define the \underline{observed Fisher info matrix}:
\[
J_n(\theta)_{jk}:=-\frac1n\sum_{i=1}^n
\frac{\partial^2}{\partial\theta_j\partial\theta_k}\log f(X_i,\theta).
\]

\noindent\underline{Lemma (last time)}\quad
If $\tilde\theta_n$ is in the line segment between $\theta_0$ \& $\hat\theta_n$,
\[
J_n(\tilde\theta_n)\xrightarrow{P}I(\theta_0).
\]
Consider a coordinate $j\in\{1,\ldots,p\}$. By the lemma \& CMT, we have
\[
\frac{\sqrt n(\hat\theta_{n,j}-\theta_{0,j})}
{\sqrt{\bigl(J_n(\hat\theta_n)^{-1}\bigr)_{jj}}}
\xrightarrow{d}N(0,1).
\]
We can use this to define the \underline{Wald confidence interval}:
\[
C_\alpha:=\left\{\theta_j\in\mathbb R:
|\theta_j-\hat\theta_{n,j}|\leq
\frac{z_\alpha}{\sqrt n}\sqrt{\bigl(J_n^{-1}(\hat\theta_n)\bigr)_{jj}}\right\},
\]
where $z_\alpha$ is s.t.\ if $Z\sim N(0,1)$,
$\mathbb P(|Z|>z_\alpha)=1-\alpha$.

We have
\[
\mathbb P_{\theta_0}(\theta_{0,j}\in C_\alpha)
\xrightarrow{n\to\infty}1-\alpha.
\]
To construct a confidence set for all of $\theta$, we use:

\medskip
\noindent\underline{Thm 5}\quad Under regularity, the \underline{Wald statistic}
\[
W(\theta_0):=n(\hat\theta_n-\theta_0)^T
J_n(\hat\theta_n)(\hat\theta_n-\theta_0)
\xrightarrow{d}\chi_p^2.
\]

\noindent\underline{Proof:}\quad By the lemma \& CMT,
$J_n(\hat\theta_n)^{1/2}\xrightarrow{P}I^{1/2}(\theta_0)$.
By Slutsky's lemma
\[
J_n^{1/2}(\hat\theta_n)\sqrt n(\hat\theta_n-\theta_0)
\xrightarrow{d}Z\sim N(0,I).
\]
By the CMT for the function $z\mapsto\|z\|^2$, we have that
\[
W(\theta_0)=\left\|J_n^{1/2}(\hat\theta_n)
\sqrt n(\hat\theta_n-\theta_0)\right\|^2
\xrightarrow{d}\|Z\|^2\sim\chi_p^2.
\]
\hfill$\square$

An asymptotic $(1-\alpha)$-confidence set for $\theta_0$ is given by the elliptical region
\[
C_\alpha:=\{\theta\in\mathbb R^p:W(\theta)<\xi_\alpha\},
\]
where $\xi_\alpha$ is s.t.\ if $W\sim\chi_p^2$,
$\mathbb P(W\geq\xi_\alpha)=\alpha$.
Again
\[
\mathbb P_{\theta_0}(\theta_0\in C_\alpha)
=\mathbb P_{\theta_0}(W(\theta_0)<\xi_\alpha)
\xrightarrow{n\to\infty}1-\alpha.
\]
Using the same Wald statistic, we can test the point null hypothesis
$H_0:\theta=\theta_0$.
Define the test
\[
\psi^{\mathrm{Wald}}=\mathbf1_{\{W(\theta_0)>\xi_\alpha\}}.
\]
By the same argument, the Type I error probability or size has
\[
\mathbb E_{H_0}\psi^{\mathrm{Wald}}\xrightarrow{n\to\infty}\alpha.
\]

\subsection*{Likelihood ratio \& score tests}

Let $H_0:\theta\in\Theta_0$, $H_1:\theta\in\Theta\subseteq\mathbb R^p$,
where $\Theta_0\subseteq\Theta$ is a linear subspace of dimension $p_0$.

\medskip
\noindent\underline{Thm 6 (Wilks' theorem)}\quad Under regularity, if $\theta_0\in\Theta_0$,
\[
\hat\Lambda_n:=2\left(
\underbrace{\sup_{\theta\in\Theta}\ell_n(\theta)}_{\ell_n(\hat\theta_n)}
-\underbrace{\sup_{\theta\in\Theta_0}\ell_n(\theta)}_{\ell_n(\theta_0)}
\right)\xrightarrow{d}\chi_{p-p_0}^2.
\]

\noindent\underline{Proof (Case $\Theta_0=\{\theta_0\}$, $p_0=0$)}

Taylor expand $\ell_n(\theta)$ around $\ell_n(\hat\theta_n)$
\[
\begin{aligned}
2\bigl(\ell_n(\theta_0)-\ell_n(\hat\theta_n)\bigr)
&=\underbrace{\frac{\partial\ell_n(\hat\theta_n)}{\partial\theta}}_{0}
(\theta_0-\hat\theta_n)\\
&\quad+\frac n2(\theta_0-\hat\theta_n)^T
J_n(\tilde\theta_n)(\theta_0-\hat\theta_n).
\end{aligned}
\]
Where $\tilde\theta_n$ is in the line segment between $\theta_0$ \& $\hat\theta_n$.
The same argument in Thm 5 is used to show that the RHS
$\xrightarrow{d}\chi_p^2$.\hfill$\square$

Wilks' test for $H_0:\theta\in\Theta_0$ is given by
\[
\psi^{\mathrm{Wilks}}=\mathbf1_{\{\hat\Lambda_n>\xi_\alpha\}},
\]
where $\xi_\alpha$ is the upper $\alpha$ point of $\chi_{p-p_0}^2$.
By Thm 6, the size
\[
\mathbb E_{\theta_0}\psi^{\mathrm{Wilks}}\xrightarrow{n\to\infty}\alpha
\qquad\forall\theta_0\in\Theta_0.
\]

\noindent\underline{Note:} This test does not require computing $J_n(\hat\theta_n)$.

The score test does not even require computing the MLE!
(For a point null hypothesis).
Define the \underline{score test statistic}
\[
\lambda_n:=\frac1n S_n(\theta_0)^T I^{-1}(\theta_0)S_n(\theta_0).
\]
Here $S_n(\theta)=\frac{\partial}{\partial\theta}\ell_n(\theta)$ is the \underline{score}.

By the Central Limit Theorem
\[
\frac1n S_n(\theta_0)\xrightarrow{d}N(0,I(\theta_0)).
\]
By the CMT $\lambda_n\xrightarrow{d}\chi_p^2$ under
\underline{$H_0:\theta=\theta_0$}.

The \underline{score test} is
\[
\psi^{\mathrm{score}}=\mathbf1_{\{\lambda_n>\xi_\alpha\}}.
\]
When $H_0:\theta\in\Theta_0$ is composite $(\dim\Theta_0>0)$,
we can define the score test statistic
\[
\lambda_n:=\frac1n S_n(\hat\theta^{(0)})^T
I^{-1}(\hat\theta^{(0)})S_n(\hat\theta^{(0)}),
\]
where $\hat\theta^{(0)}$ is a ``restricted'' MLE, that is a maximiser of
$\ell_n$ over $\theta\in\Theta_0$.
One can show that under regularity,
$\lambda_n\xrightarrow{d}\chi_{p-p_0}^2$ as $n\to\infty$ if $\theta_0\in\Theta_0$.

\subsection*{6. The Delta Method}

Let $\hat\theta$ be a MLE. We wish to estimate $\phi(\theta_0)$ for some
(possibly non-linear, non-injective) function $\phi:\Theta\to\mathbb R^m$.
A \underline{profile likelihood estimator} of $\phi(\theta_0)$ maximises
\[
L^*(\psi)=\sup_{\substack{\theta\in\Theta\\\phi(\theta)=\psi}}L(\theta)
\]
over $\psi\in\phi(\Theta)$.

\medskip
\noindent\underline{Lemma}\quad The plug-in MLE, $\phi(\hat\theta)$,
maximises $L^*$.

\noindent\underline{Proof}\quad Suppose $\exists\,\tilde\psi$ s.t.
\[
L^*(\tilde\psi)>L^*(\phi(\hat\theta))+\varepsilon
\qquad\text{for some }\varepsilon>0.
\]
By definition of $L^*$, $\exists\,\tilde\theta$ s.t.
\[
L(\tilde\theta)>L^*(\tilde\psi)-\varepsilon
>L^*(\phi(\hat\theta))\geq L(\hat\theta),
\]
which contradicts the maximality of the MLE $\hat\theta$.\hfill$\square$

The Delta Method allows us to characterise the asymptotic distribution of
fluctuations $(\phi(\hat\theta)-\phi(\theta_0))\sqrt n$, which can be used
for inference of $\phi(\theta_0)$.

\medskip
\noindent\underline{Theorem 7}\quad Let $\hat\theta_n$ be an estimator s.t.
\[
r_n(\hat\theta_n-\theta_0)\xrightarrow{d}Z
\]
for some random variable $Z$ and a deterministic rate $r_n\xrightarrow{n\to\infty}\infty$
(e.g.\ $r_n=\sqrt n$). [$\hat\theta_n$ need not be the MLE!]
Let $\phi:\Theta\to\mathbb R^m$ be a continuously differentiable function. Then
\[
r_n\bigl(\phi(\hat\theta_n)-\phi(\theta_0)\bigr)
\xrightarrow{d}
\begin{pmatrix}
\nabla\phi_1(\theta_0)^T\\
\vdots\\
\nabla\phi_m(\theta_0)^T
\end{pmatrix}Z
\qquad\text{as }n\to\infty.
\]

\noindent\underline{Proof}\quad By the mean value theorem,
\[
r_n\bigl(\phi(\hat\theta_n)-\phi(\theta_0)\bigr)
=\underbrace{\begin{pmatrix}
\nabla\phi_1(\tilde\theta_n^{(1)})^T\\
\vdots\\
\nabla\phi_m(\tilde\theta_n^{(m)})^T
\end{pmatrix}}_{=:D_n}
\underbrace{(\hat\theta_n-\theta_0)r_n}_{\xrightarrow{d}Z},
\]
where $\tilde\theta_n^{(i)}$ is on the segment between $\theta_0$ and
$\hat\theta_n$ for each $i=1,\ldots,m$.
Note that $r_n(\hat\theta_n-\theta_0)=O_{\mathbb P}(1)$ (by Prop 1(4));
as $r_n\to\infty$, this shows that $\hat\theta_n\xrightarrow{P}\theta_0$
(Ex Sheet). Therefore $\tilde\theta_n^{(i)}\xrightarrow{P}\theta_0$
for all $i=1,\ldots,m$.
As $\phi$ is continuously differentiable, by the CMT we have
$D_n\xrightarrow{P}\nabla\phi(\theta_0)$.
Finally, Slutsky's lemma gives the desired conclusion.\hfill$\square$

\medskip
\noindent\underline{Notation (supplement):}\quad
Here the board's $\nabla\phi(\theta_0)$ denotes the Jacobian matrix
\[
J_\phi(\theta_0):=
\begin{pmatrix}
\nabla\phi_1(\theta_0)^T\\
\vdots\\
\nabla\phi_m(\theta_0)^T
\end{pmatrix},
\]
with one transposed component gradient per row, just as in $D_n$
(with $k$ outputs and $m$ input coordinates, the supplement writes these
matrices as elements of $\mathbb R^{k\times m}$).

\medskip
\noindent\underline{Corollary}\quad If $\hat\theta_n$ is an asymptotically normal MLE,
\[
\sqrt n(\hat\theta_n-\theta_0)\xrightarrow{d}
N(0,I^{-1}(\theta_0)),
\]
then
\[
\sqrt n\bigl(\phi(\hat\theta_n)-\phi(\theta_0)\bigr)
\xrightarrow{d}N\bigl(0,
\nabla\phi(\theta_0)^T I^{-1}(\theta_0)\nabla\phi(\theta_0)\bigr).
\]
Recall that by the Cram\'er--Rao lower bound for $\phi:\Theta\to\mathbb R$,
any unbiased estimate $\hat\phi$
($\mathbb E_{\theta_0}\hat\phi=\phi(\theta_0)$ for all $\theta_0$) has
\[
\operatorname{Var}(\hat\phi)\geq\frac1n
\nabla\phi(\theta_0)^T I^{-1}(\theta_0)\nabla\phi(\theta_0)
\qquad\text{for any finite }n.
\]
We see that the plug-in MLE $\phi(\hat\theta_n)$ attains or matches this
lower bound asymptotically.

\medskip
\noindent\underline{Notation (supplement):}\quad
The board's formulas above are retained: their $\nabla\phi(\theta_0)$
is the transpose of the Jacobian $J_\phi(\theta_0)$ defined in the proof,
so the covariance is $J_\phi I^{-1}J_\phi^T$.
For scalar $\phi$, this is the column gradient, explaining the transpose
on the left and its absence on the right in the Cram\'er--Rao bound.

\medskip
\noindent\underline{Example}\quad Suppose $(X_i)\overset{\mathrm{iid}}{\sim}N(\mu,1)$.
We wish to estimate $\mu_0^2$ with the plug-in MLE $(\bar X_n)^2$.
In this case we know
\[
\sqrt n(\bar X_n-\mu_0)\xrightarrow{d}N(0,1)
\qquad\text{by the CLT}.
\]
By Theorem 7, we have
\[
\sqrt n(\bar X_n^2-\mu_0^2)\xrightarrow{d}N(0,4\mu_0^2).
\]
When $\mu_0=0$, the limit has variance $0$
($\bar X_n^2-\mu_0^2$ converges ``faster'' than $\sqrt n$).
We can instead scale the fluctuations $\bar X_n^2-\mu_0^2$ by $r_n=n$:
\[
n(\bar X_n^2-0)
=\bigl(\underbrace{\sqrt n\,\bar X_n}_{\xrightarrow{d}N(0,1)\text{ by CLT}}\bigr)^2
\xrightarrow{d}\chi_1^2
\qquad\text{by continuous mapping}.
\]

\section*{II. BAYESIAN INFERENCE}

Suppose observation $X$ comes from a statistical model
$\{f(\cdot,\theta):\theta\in\Theta\}$ on a sample space $\mathcal X$.
We assign a \underline{prior distribution} $\pi$ on $\Theta$, which expresses
subjective opinions about $\theta$.
Then we consider $X,\theta$ to be random variables with joint pdf (pmf)
\[
\pi(\theta)f(x,\theta)\qquad\text{on }\mathcal X\times\Theta,
\]
i.e.\ $\theta\sim\pi$, and $X\mid\theta\sim f(\cdot,\theta)$.
The \underline{posterior distribution} is the conditional distribution of
$\theta\mid X$, with pdf
\[
\pi(\theta\mid x)=
\frac{\pi(\theta)f(x,\theta)}
{\displaystyle\int_\Theta\pi(\theta)f(x,\theta)\,d\theta}
\qquad\text{``Bayes' rule''}.
\]

\noindent\underline{Notation (supplement):}\quad
The board's $f(x,\theta)$ means $f(x\mid\theta)$, the conditional pdf
(or pmf) of $X$ given $\theta$; the joint pdf (or pmf) is
$\pi(\theta)f(x,\theta)$.

When $X=(X_1,\ldots,X_n)$ are iid and $f(\cdot,\theta)$ denotes the pdf
of a single observation, then
\[
\pi(\theta\mid x_1,\ldots,x_n)=
\frac{\displaystyle\pi(\theta)\prod_{i=1}^n f(x_i,\theta)}
{\displaystyle\int_\Theta\pi(\theta)\prod_{i=1}^n f(x_i,\theta)\,d\theta}.
\]
The posterior can be used to make decisions under uncertainty
\underline{conditional} on a fixed observation.
We use the principle of maximising expected utility
(see von Neumann--Morgenstern theorem).
This motivates methods for:
\begin{enumerate}
\item Estimation, e.g.\ let
\[
\hat\theta_n=\int_\Theta\theta\,\pi(\theta\mid x_1,\ldots,x_n)\,d\theta.
\]
\item Uncertainty quantification, e.g.\ a \underline{credible set} $\hat C_n$:
\[
\int_{\hat C_n}\pi(\theta\mid x_1,\ldots,x_n)\,d\theta=1-\alpha.
\tag{$*$}
\]
\end{enumerate}
For example, if $\Theta=\mathbb R$, let
\[
\hat C_n:=[\hat\theta_n-R_n,\,\hat\theta_n+R_n],
\]
with $R_n$ chosen s.t.\ $(*)$ holds.
Note that $\hat\theta_n,\hat C_n$ are simply functions of $x_1,\ldots,x_n$.
But we can regard these as random functions of
$X_1,\ldots,X_n\sim P_{\theta_0}$ coming from the model with a true $\theta_0$.
