\section{Martingales}

We work with a probability space $(\Omega,\mathcal{F},\mathbb{P})$.

\subsection{Definitions}

A \emph{filtration} is a sequence of $\sigma$-algebras
$(\mathcal{F}_n)_{n\geq0}$ such that
\[
\mathcal{F}_n\subseteq\mathcal{F}_{n+1}\subseteq\mathcal{F}.
\]

A \emph{random process} is a sequence
$X=(X_n)_{n\geq0}$ of random variables.

Given a filtration $(\mathcal{F}_n)_{n\geq0}$ on
$(\Omega,\mathcal{F},\mathbb{P})$, we say that $X$ is a
\emph{martingale} if:
\begin{enumerate}
    \item[(a)] $X$ is adapted, i.e.\ $X_n$ is
    $\mathcal{F}_n$-measurable for all $n$.

    \item[(b)] $X$ is integrable, i.e.\ $X_n$ is integrable
    for all $n$:
    \[
    \mathbb{E}[|X_n|]<\infty.
    \]

    \item[(c)] $X$ satisfies the martingale property:
    \[
    \mathbb{E}[X_{n+1}\mid\mathcal{F}_n]
    =
    X_n
    \qquad\text{a.s., for all }n.
    \]
\end{enumerate}

Given (a) and (b), condition (c) is equivalent to
\[
\mathbb{E}[X_{n+1}\mathbf{1}_A]
=
\mathbb{E}[X_n\mathbf{1}_A]
\qquad
\text{for all }n
\text{ and all }A\in\mathcal{F}_n.
\]

If (c) is replaced by
\[
\mathbb{E}[X_{n+1}\mid\mathcal{F}_n]
\geq X_n
\qquad\text{a.s., for all }n,
\]
or, equivalently,
\[
\mathbb{E}[X_{n+1}\mathbf{1}_A]
\geq
\mathbb{E}[X_n\mathbf{1}_A]
\qquad
\text{for all }n
\text{ and all }A\in\mathcal{F}_n,
\]
then $(X_n)_{n\geq0}$ is a \emph{submartingale}.

If (c) is replaced by
\[
\mathbb{E}[X_{n+1}\mid\mathcal{F}_n]
\leq X_n
\qquad\text{a.s., for all }n,
\]
or, equivalently,
\[
\mathbb{E}[X_{n+1}\mathbf{1}_A]
\leq
\mathbb{E}[X_n\mathbf{1}_A]
\qquad
\text{for all }n
\text{ and all }A\in\mathcal{F}_n,
\]
then $(X_n)_{n\geq0}$ is a \emph{supermartingale}.

\subsection{Natural filtration}

Given a random process $X=(X_n)_{n\geq0}$, its
\emph{natural filtration} $(\mathcal{F}_n^X)_{n\geq0}$
is defined by
\[
\begin{aligned}
\mathcal{F}_n^X
&=
\sigma(X_0,X_1,\ldots,X_n)\\
&=
\left\{
\{(X_0,X_1,\ldots,X_n)\in B\}:
B\in\mathcal{B}(\mathbb{R}^{n+1})
\right\}.
\end{aligned}
\]
This is the smallest filtration making $X$ adapted.

\paragraph{Exercise.}
Show that a martingale $X$ with respect to
$(\mathcal{F}_n)_{n\geq0}$ is also a martingale with respect
to $(\mathcal{F}_n^X)_{n\geq0}$.

\subsection{Examples}

Let $X_1,X_2,\ldots$ be a sequence of independent integrable
random variables. Set
\[
\mathcal{F}_n=\mathcal{F}_n^X.
\]
Define
\[
S_0=0,
\qquad
Z_0=1,
\]
and
\[
S_n=X_1+\cdots+X_n,
\qquad
Z_n=\prod_{i=1}^{n}X_i.
\]

\begin{itemize}
    \item If $\mathbb{E}[X_n]=0$ for all $n$, then
    $(S_n)_{n\geq0}$ is a martingale.

    \item If $X_n\geq0$ and $\mathbb{E}[X_n]=1$ for all $n$,
    then $(Z_n)_{n\geq0}$ is a martingale.
\end{itemize}

To see this, take $A\in\mathcal{F}_n$. Then
\[
\begin{aligned}
\mathbb{E}[S_{n+1}\mathbf{1}_A]
&=
\mathbb{E}[X_{n+1}\mathbf{1}_A]
+
\mathbb{E}[S_n\mathbf{1}_A]\\
&=
\underbrace{
\mathbb{E}[X_{n+1}]\mathbb{P}(A)
}_{=0}
+
\mathbb{E}[S_n\mathbf{1}_A]\\
&=
\mathbb{E}[S_n\mathbf{1}_A].
\end{aligned}
\]

Also,
\[
\begin{aligned}
\mathbb{E}[Z_{n+1}\mathbf{1}_A]
&=
\mathbb{E}[X_{n+1}Z_n\mathbf{1}_A]\\
&=
\underbrace{
\mathbb{E}[X_{n+1}]
}_{=1}
\mathbb{E}[Z_n\mathbf{1}_A]\\
&=
\mathbb{E}[Z_n\mathbf{1}_A].
\end{aligned}
\]
This also shows integrability.

\section{Conditional Probabilities and Expectation}

For $\mathbb{P}(B)>0$,
\[
\mathbb{P}(A\mid B)
=
\frac{\mathbb{P}(A\cap B)}{\mathbb{P}(B)},
\]
and
\[
\mathbb{E}[X\mid B]
=
\frac{\mathbb{E}[X\mathbf{1}_B]}{\mathbb{P}(B)}.
\]

\subsection{Existence and uniqueness}

\paragraph{Theorem.}
Let $\mathcal{G}\subseteq\mathcal{F}$ be a $\sigma$-algebra.
Let $X$ be an integrable random variable. Then there exists
a random variable $Y$ such that:
\begin{enumerate}
    \item[(a)] $Y$ is $\mathcal{G}$-measurable;

    \item[(b)]
    \[
    \mathbb{E}[X\mathbf{1}_A]
    =
    \mathbb{E}[Y\mathbf{1}_A]
    \qquad\text{for all }A\in\mathcal{G}.
    \]
\end{enumerate}
Moreover, if $Y'$ is another random variable satisfying
(a) and (b), then
\[
Y=Y'\qquad\text{a.s.}
\]

\paragraph{Proof of uniqueness.}
Let $X'$ be another integrable random variable with
$X\leq X'$. Suppose $Y'$ satisfies (a) and (b) with respect
to $X'$.

Set
\[
A=\{Y>Y'\},
\qquad
Z=(Y-Y')\mathbf{1}_A.
\]
Then $A\in\mathcal{G}$ and $Z\geq0$. Hence
\[
\begin{aligned}
\mathbb{E}[Z]
&=
\mathbb{E}[(Y-Y')\mathbf{1}_A]\\
&=
\mathbb{E}[Y\mathbf{1}_A]
-
\mathbb{E}[Y'\mathbf{1}_A]\\
&=
\mathbb{E}[X\mathbf{1}_A]
-
\mathbb{E}[X'\mathbf{1}_A]\\
&=
\mathbb{E}[(X-X')\mathbf{1}_A]
\leq0.
\end{aligned}
\]
Therefore $Z=0$ a.s., so
\[
Y\leq Y'\qquad\text{a.s.}
\]
Taking $X'=X$, we also obtain $Y\geq Y'$ a.s. Thus
\[
Y=Y'\qquad\text{a.s.}
\]

\section*{Optional stopping}

We say that a random time
\[
T:\Omega\longrightarrow \{0,1,2,\ldots\}\cup\{\infty\}
\]
is a \textbf{stopping time} if
\[
\{T\leq n\}\in\mathcal{F}_n
\qquad\text{for all }n.
\]

\noindent\textbf{Example.}
Let $(X_n)_{n\geq 0}$ be an adapted process, with values in $\mathbb{Z}$, say.
Set
\[
T=\inf\{n\geq 0:X_n=0\}.
\]
Then
\[
\{T\leq n\}
=
\underbrace{\{X_0=0\}}_{\in\mathcal{F}_0}
\cup
\underbrace{\{X_1=0\}}_{\in\mathcal{F}_1}
\cup\cdots\cup
\underbrace{\{X_n=0\}}_{\in\mathcal{F}_n},
\]
with
\[
\mathcal{F}_0\subseteq\mathcal{F}_1
\subseteq\cdots\subseteq\mathcal{F}_n.
\]
So $T$ is a stopping time.

\subsection*{Optional stopping theorem}

Let $(M_n)_{n\geq 0}$ be a martingale and let $T$ be a bounded stopping time.
Then
\[
\mathbb{E}(M_T)=\mathbb{E}(M_0),
\qquad
\bigl(M_T(\omega)=M_{T(\omega)}(\omega)\bigr).
\]

\noindent\textbf{Proof.}
Choose $n$ such that $T\leq n$. Then
\[
\begin{aligned}
M_T
&=M_0+(M_1-M_0)+\cdots+(M_T-M_{T-1})\\
&=M_0+\sum_{k=1}^{n}(M_k-M_{k-1})
  \mathbf{1}_{\{T\geq k\}}.
\end{aligned}
\]
Now
\[
\{T\geq k\}
=
\{T\leq k-1\}^{c}
\in\mathcal{F}_{k-1},
\]
so
\[
\mathbb{E}\!\left(
(M_k-M_{k-1})\mathbf{1}_{\{T\geq k\}}
\right)=0
\qquad\text{for all }k.
\]
Take expectations to see
\[
\mathbb{E}(M_T)=\mathbb{E}(M_0)+0.
\qquad\square
\]

\subsection*{OST II}

Let $(M_n)_{n\geq 0}$ be a martingale and let $T$ be a finite-valued stopping time.
Suppose there exists $C<\infty$ such that one of the following holds:
\begin{enumerate}
    \item[(a)]
    \[
    |M_n|\leq C
    \qquad\text{for all }n\leq T.
    \]

    \item[(b)]
    \[
    \mathbb{E}(T)<\infty
    \qquad\text{and}\qquad
    |M_n-M_{n-1}|\leq C
    \qquad\text{for all }n\leq T.
    \]
\end{enumerate}
Then
\[
\mathbb{E}(M_T)=\mathbb{E}(M_0).
\]

\noindent\textbf{Proof.}
Define
\[
T_n=T\wedge n.
\]
Then $T_n$ is a bounded stopping time.
Since $T<\infty$, $T_n=T$ eventually, so
\[
M_{T_n}\longrightarrow M_T
\qquad\text{as }n\longrightarrow\infty.
\]
By optional stopping, we have
\[
\mathbb{E}(M_{T_n})=\mathbb{E}(M_0).
\]

If (a) holds, then
\[
\sup_n |M_{T_n}|\leq C,
\]
so
\[
\mathbb{E}(M_{T_n})\longrightarrow\mathbb{E}(M_T)
\]
by bounded convergence.

If (b) holds, then
\[
\sup_n |M_{T_n}|\leq CT+|M_0|
\qquad\text{and}\qquad
\mathbb{E}(T)<\infty,
\]
so
\[
\mathbb{E}(M_{T_n})\longrightarrow\mathbb{E}(M_T)
\]
by dominated convergence.

In any case, we can let $n\longrightarrow\infty$ to see
\[
\mathbb{E}(M_T)=\mathbb{E}(M_0).
\qquad\square
\]

\subsection*{Examples}

Let $X_1,X_2,\ldots$ and $Y_1,Y_2,\ldots$ be independent random variables such that
\[
\mathbb{P}(X_n=1)=\mathbb{P}(X_n=-1)=\frac12,
\qquad
\mathbb{P}(Y_n=0)=\mathbb{P}(Y_n=2)=\frac12.
\]
Set
\[
S_n=1+X_1+\cdots+X_n,
\qquad
Z_n=\prod_{i=1}^{n}Y_i,
\qquad
(Z_0=1).
\]
Then $(S_n)_{n\geq 0}$ and $(Z_n)_{n\geq 0}$ are martingales.

Set
\[
T=\inf\{n\geq 0:S_n=0\}.
\]
Then
\[
\mathbb{P}(T<\infty)=1,
\]
but
\[
\mathbb{E}(S_0)=1,
\qquad
\mathbb{E}(S_T)=0.
\]

Now set
\[
T=\inf\{n\geq 0:Z_n=0\}.
\]
Then
\[
\mathbb{P}(T\leq n)=1-2^{-n},
\]
so
\[
\mathbb{P}(T<\infty)=1.
\]
In fact,
\[
\mathbb{E}(T)<\infty.
\]
But
\[
\mathbb{E}(Z_T)=0,
\qquad
\mathbb{E}(Z_0)=1.
\]

Y=\mathbb{E}(X\mid\mathcal{G})

\section*{Properties of conditional expectation}

\begin{enumerate}
    \item[(i)]
    \[
    \mathbb{E}\!\left(\mathbb{E}(X\mid\mathcal{G})\right)
    =
    \mathbb{E}(X).
    \]

    \item[(ii)]
    If $X$ is $\mathcal{G}$-measurable, then
    \[
    \mathbb{E}(X\mid\mathcal{G})=X
    \qquad\text{a.s.}
    \]

    \item[(iii)]
    If $X$ is independent of $\mathcal{G}$, then
    \[
    \mathbb{E}(X\mid\mathcal{G})=\mathbb{E}(X)
    \qquad\text{a.s.}
    \]

    \item[(iv)]
    If $X\geq 0$ a.s., then
    \[
    \mathbb{E}(X\mid\mathcal{G})\geq 0
    \qquad\text{a.s.}
    \]

    \item[(v)]
    \textbf{Linearity.}
    For $W,X$ integrable and $\alpha,\beta\in\mathbb{R}$,
    \[
    \mathbb{E}(\alpha X+\beta W\mid\mathcal{G})
    =
    \alpha\mathbb{E}(X\mid\mathcal{G})
    +
    \beta\mathbb{E}(W\mid\mathcal{G}).
    \]
\end{enumerate}

Take
\[
Y=\mathbb{E}(X\mid\mathcal{G}),
\qquad
U=\mathbb{E}(W\mid\mathcal{G}).
\]
Then $\alpha Y+\beta U$ is $\mathcal{G}$-measurable and integrable.

For $A\in\mathcal{G}$,
\[
\begin{aligned}
\mathbb{E}\!\left((\alpha X+\beta W)\mathbf{1}_A\right)
&=\alpha\mathbb{E}(X\mathbf{1}_A)
  +\beta\mathbb{E}(W\mathbf{1}_A)\\
&=\alpha\mathbb{E}(Y\mathbf{1}_A)
  +\beta\mathbb{E}(U\mathbf{1}_A)\\
&=\mathbb{E}\!\left((\alpha Y+\beta U)\mathbf{1}_A\right).
\end{aligned}
\]
Hence
\[
\alpha Y+\beta U
=
\mathbb{E}(\alpha X+\beta W\mid\mathcal{G})
\qquad\text{a.s.}
\]

\subsection*{Proposition (Tower property)}

Let $\mathcal{H}$ be another $\sigma$-algebra, with
\[
\mathcal{H}\subseteq\mathcal{G}.
\]
Then
\[
\mathbb{E}\!\left(
\mathbb{E}(X\mid\mathcal{G})\mid\mathcal{H}
\right)
=
\mathbb{E}(X\mid\mathcal{H}).
\]

\noindent\textbf{Proof.}
Take
\[
Y=
\mathbb{E}\!\left(
\mathbb{E}(X\mid\mathcal{G})\mid\mathcal{H}
\right)
\qquad\text{a.s.}
\]
Then, for $A\in\mathcal{H}$, we have $A\in\mathcal{G}$, so
\[
\mathbb{E}(Y\mathbf{1}_A)
\underset{A\in\mathcal{H}}{=}
\mathbb{E}\!\left(
\mathbb{E}(X\mid\mathcal{G})\mathbf{1}_A
\right)
\underset{A\in\mathcal{G}}{=}
\mathbb{E}(X\mathbf{1}_A).
\]
$Y$ is integrable and $\mathcal{H}$-measurable, so
\[
Y=\mathbb{E}(X\mid\mathcal{H})
\qquad\text{a.s.}
\qquad\square
\]

\subsection*{Properties of conditional expectation (continued)}

\noindent\textbf{Proposition (Taking out what is known).}
Let $\mathcal{G}\subseteq\mathcal{F}$ be a $\sigma$-algebra, and let $X$ be an integrable random variable. Suppose that $Z$ is a bounded $\mathcal{G}$-measurable random variable. Then
\[
\mathbb{E}[ZX\mid\mathcal{G}]
=
Z\mathbb{E}[X\mid\mathcal{G}]
\qquad \text{a.s.}
\]

\noindent\textbf{Proof.}
Take $Y=\mathbb{E}[X\mid\mathcal{G}]$ a.s. Note that $ZY$ is integrable and $\mathcal{G}$-measurable.

Take $A\in\mathcal{G}$. We will show that
\[
\mathbb{E}[ZX\mathbf{1}_A]
=
\mathbb{E}[ZY\mathbf{1}_A],
\]
so that $ZY$ satisfies (a) and (b) for $ZX$, and we are done.

Consider first the case $Z=\mathbf{1}_B$, where $B\in\mathcal{G}$. Then $A\cap B\in\mathcal{G}$, so
\[
\begin{aligned}
\mathbb{E}[ZX\mathbf{1}_A]
&=\mathbb{E}[X\mathbf{1}_{A\cap B}]\\
&=\mathbb{E}[Y\mathbf{1}_{A\cap B}]\\
&=\mathbb{E}[ZY\mathbf{1}_A].
\end{aligned}
\]
This extends to simple random variables
\[
Z=\sum_{i=1}^{n}a_i\mathbf{1}_{B_i}
\]
by linearity.

For $Z\geq 0$ and $X\geq 0$, consider
\[
Z_n
=
\left(2^{-n}\left\lfloor 2^n Z\right\rfloor\right)\wedge n.
\]
Then $Z_n$ is simple, so
\[
\mathbb{E}[Z_nX\mathbf{1}_A]
=
\mathbb{E}[Z_nY\mathbf{1}_A].
\]
Since $X\geq 0$, we have $Y\geq 0$ a.s., so
\[
Z_nX\uparrow ZX,
\qquad
Z_nY\uparrow ZY.
\]
Letting $n\to\infty$,
\[
\begin{aligned}
\mathbb{E}[Z_nX\mathbf{1}_A]
&\longrightarrow \mathbb{E}[ZX\mathbf{1}_A],\\
\mathbb{E}[Z_nY\mathbf{1}_A]
&\longrightarrow \mathbb{E}[ZY\mathbf{1}_A].
\end{aligned}
\]

Finally, for general $X$ and $Z$, use
\[
X=X^+-X^-,
\qquad
Z=Z^+-Z^-,
\]
and linearity to obtain the desired formula.
\hfill$\square$

\medskip

\noindent\textbf{Theorem.}
Let $M$ be a martingale and let $T$ be a stopping time. Consider the stopped process
\[
M_n^T:=M_{T\wedge n}.
\]
Then $M^T$ is also a martingale.

\medskip

\noindent\textbf{Proof.}
We have
\[
M_n^T
=
M_0+\sum_{k=1}^{n}(M_k-M_{k-1})\mathbf{1}_{\{k\leq T\}}.
\]
Since $\{k\leq T\}\in\mathcal{F}_k$, this shows that $M_n^T$ is $\mathcal{F}_n$-measurable. Also, $M^T$ is integrable.

To show the martingale property, it suffices to show, for $A\in\mathcal{F}_n$ and all $n$, that
\[
\mathbb{E}\left[(M_{n+1}^T-M_n^T)\mathbf{1}_A\right]=0.
\]
But
\[
M_{n+1}^T-M_n^T
=
(M_{n+1}-M_n)\mathbf{1}_{\{n+1\leq T\}}.
\]
Note that
\[
\{n+1\leq T\}
=
\{T\leq n\}^{c}
\in\mathcal{F}_n.
\]
So
\[
\mathbb{E}\left[
(M_{n+1}-M_n)
\underbrace{
\mathbf{1}_{\{n+1\leq T\}}\mathbf{1}_A
}_{\mathcal{F}_n\text{-measurable}}
\right]
=0,
\]
as required. Note that in the final step we used tower rule and "take out what is known".
\hfill$\square$

\subsection*{Martingale transforms}

Say that a process $(H_n)_{n\geq 1}$ is \emph{previsible} if $H_n$ is $\mathcal{F}_{n-1}$-measurable for all $n$.

\medskip

\noindent\textbf{Theorem.}
Let $M$ be a martingale and let $H$ be a bounded previsible process. We define the transform $H\cdot M$ of $M$ by $H$ by
\[
(H\cdot M)_n
=
Y_n
=
\sum_{k=1}^{n}H_k(M_k-M_{k-1}),
\qquad n\geq 1,
\]
and
\[
(H\cdot M)_0=Y_0=0.
\]

\medskip

\noindent\textbf{Interpretation.}
Think of $M_n$ as the price of an asset at time $n$.

An investor holds $H_n$ units of the asset between time $n-1$ and time $n$.

Then the profit made from time $0$ to time $n$ is $Y_n$.

The previsible restriction is natural.

\medskip

Then $H\cdot M$ is also a martingale. In particular,
\[
\mathbb{E}[(H\cdot M)_n]
=
\mathbb{E}[(H\cdot M)_0]
=
0.
\]

\noindent\textbf{Proof.}
Note that
\[
(H\cdot M)_{n+1}-(H\cdot M)_n
=
H_{n+1}(M_{n+1}-M_n),
\]
and, by taking out what is known,
\[
\begin{aligned}
&\mathbb{E}\left[
H_{n+1}(M_{n+1}-M_n)\mid\mathcal{F}_n
\right]\\
&\qquad=
H_{n+1}
\underbrace{
\mathbb{E}[M_{n+1}-M_n\mid\mathcal{F}_n]
}_{=\,0\text{, as }M\text{ is a martingale}}.
\end{aligned}
\]
So $H\cdot M$ is a martingale, as claimed.
\hfill$\square$

\section*{2. Pricing contingent claims}

\subsection*{2.1 Multi-period asset price model}

Let $(\bar S_n)_{0\leq n\leq T}$ be a random process in
$\mathbb{R}^{d+1}$, with
\[
\bar S_n=(S_n^0,S_n^1,\ldots,S_n^d)=(S_n^0,S_n),
\qquad S_n\in\mathbb{R}^d.
\]
We say that $(S_n^0)_{0\leq n\leq T}$ is a \emph{numeraire} if
$S_n^0>0$ for all $n$.

If $(S_n^0)_{0\leq n\leq T}$ is a numeraire, we call
$(\bar S_n)_{0\leq n\leq T}$ a \emph{$T$-period asset price model}.
If $T=1$, it is called a single-period model.

We think of $(S_n^0)_{0\leq n\leq T}$ as a bank account or bond, and write
\[
S_n^0=(1+r_n)S_{n-1}^0,
\]
where $r_n$ is the interest rate.
There are $T$ time periods:
\[
0\longrightarrow1,\quad 1\longrightarrow2,\quad\ldots,\quad
T-1\longrightarrow T.
\]
We think of $S_n^i$ as the price of one unit of asset $i$ at time $n$,
for $i=1,\ldots,d$.

We will focus on \emph{discounted prices}:
\[
\bar X_n=\frac{\bar S_n}{S_n^0}
=(1,X_n^1,\ldots,X_n^d),
\qquad
X_n^i=\frac{S_n^i}{S_n^0}.
\]
To determine the model fully, we could suppose that the interest rates
$r_n$ are non-random and that prices take discrete values. We would then
want to know the values of
\[
\mathbb{P}(S_0=s_0,S_1=s_1,\ldots,S_T=s_T)
\]
for all possible sequences $(s_n)_{0\leq n\leq T}$ in $\mathbb{R}^d$.

\subsubsection*{Portfolios}

Let $(\bar\Theta_n)_{1\leq n\leq T}$ be a previsible random process in
$\mathbb{R}^{d+1}$, so that $\bar\Theta_n$ is
$\mathcal{F}_{n-1}$-measurable.
We use $(\bar\Theta_n)_{1\leq n\leq T}$ to describe a \emph{portfolio}
in which we hold $\Theta_n^i$ units of asset $i$ during the $n$th period,
from $n-1$ to $n$.

We say that $(\bar\Theta_n)_{1\leq n\leq T}$ is \emph{self-financing} if
\[
\underbrace{\bar\Theta_n\cdot\bar S_n}_{\text{value at the end of period }n}
=
\underbrace{\bar\Theta_{n+1}\cdot\bar S_n}_{\text{value at the start of period }n+1},
\qquad n=1,\ldots,T-1.
\]

The \emph{value function} of the portfolio, $(V_n)_{0\leq n\leq T}$,
is given by
\[
V_0=\bar\Theta_1\cdot\bar X_0,
\qquad
V_n=\bar\Theta_n\cdot\bar X_n
=\sum_{i=0}^{d}\Theta_n^i X_n^i,
\qquad n=1,\ldots,T.
\]

\medskip

\noindent\textbf{Proposition.}
Let $V_0\in\mathbb{R}$ and let $(\Theta_n)_{1\leq n\leq T}$ be a
previsible process in $\mathbb{R}^d$.
Then there exists a previsible process $(\Theta_n^0)_{1\leq n\leq T}$
such that $(\bar\Theta_n)_{1\leq n\leq T}$ is a self-financing portfolio
with initial value $V_0$, where $\bar\Theta_n=(\Theta_n^0,\Theta_n)$.
Moreover,
\[
V_m=V_0+\sum_{n=1}^{m}\Theta_n\cdot(X_n-X_{n-1}),
\qquad 0\leq m\leq T.
\]

\medskip

\noindent\textbf{Proof.}
We have to satisfy the conditions
\[
V_0=\bar\Theta_1\cdot\bar X_0
=\Theta_1^0+\Theta_1\cdot X_0
\]
and
\[
\begin{aligned}
\bar\Theta_n\cdot\bar S_n
&=\bar\Theta_{n+1}\cdot\bar S_n,\\
\Theta_n^0S_n^0+\Theta_n\cdot S_n
&=\Theta_{n+1}^0S_n^0+\Theta_{n+1}\cdot S_n,
\qquad n=1,\ldots,T-1.
\end{aligned}
\]
We can solve these uniquely for a previsible process
$(\Theta_n^0)_{1\leq n\leq T}$.

To obtain the formula for $V_m$, note that the change in discounted value
in period $n$ is
\[
\sum_{i=1}^{d}\Theta_n^i(X_n^i-X_{n-1}^i)
=\Theta_n\cdot(X_n-X_{n-1}).
\]
The formula follows by induction.
\hfill$\square$

\subsection*{2.2 Contingent claims}

A \emph{(European) contingent claim} $C$ with maturity $T$ is a
non-negative $\mathcal{F}_T$-measurable random variable.
We think of $C$ as a contract which pays the investor $C$ at time $T$.
The term ``European'' signifies that $T$ is fixed.

Here we are thinking of
\[
\mathcal{F}_T=\sigma(S_0,S_1,\ldots,S_T).
\]

\subsubsection*{Examples}

Restrict to $d=1$, so there is one risky asset and $S_n\in\mathbb{R}$.

\begin{enumerate}
    \item[(a)] $(S_T-K)^+$ is the payoff of a \emph{call} with strike
    price $K$ and maturity $T$. It gives the holder the right, but not the
    obligation, to buy one unit of the asset at time $T$ for price $K$.

    \item[(b)] $(S_T-K)^-$ is the payoff of a \emph{put} with strike
    price $K$ and maturity $T$. This is the same as a call, but for
    selling at price $K$.
\end{enumerate}

\subsection*{2.3 Equivalent probability measures}

\[
(\Omega,\mathcal F,\mathbb P).
\]
Say that a prob.\ meas.\ $\widetilde{\mathbb P}$ on $(\Omega,\mathcal F)$
has \emph{density} $\rho$ w.r.\ to $\mathbb P$ if $\rho$ is a random variable,
$\rho\geq0$ and
\[
\widetilde{\mathbb P}(A)=\mathbb E(\rho\mathbf1_A),
\qquad A\in\mathcal F.
\]
By standard arguments in this case
\[
\widetilde{\mathbb E}(X)=\mathbb E(\rho X),
\qquad\text{any }X\geq0\text{ r.v.}
\]

In the case where $\rho$ takes values in $(0,\infty)$ then $\mathbb P$
has density $1/\rho$ w.r.\ to $\widetilde{\mathbb P}$,
\[
\mathbb P(A)=\mathbb E\left(\rho\,\frac{\mathbf1_A}{\rho}\right)
=\widetilde{\mathbb E}\left(\frac1\rho\mathbf1_A\right).
\]
We write
\[
\rho=\frac{d\widetilde{\mathbb P}}{d\mathbb P}\quad\text{a.s.},
\qquad
\frac1\rho=\frac{d\mathbb P}{d\widetilde{\mathbb P}}\quad\text{a.s.}
\]
Suppose $\rho,\rho'$ are both densities for $\widetilde{\mathbb P}$
w.r.\ to $\mathbb P$, then $\rho=\rho'$ a.s.

Say that $\mathbb P,\widetilde{\mathbb P}$ are \emph{equivalent} if
$d\widetilde{\mathbb P}/d\mathbb P$ is a r.v.\ $\rho$ with values in $(0,\infty)$.

\subsection*{2.4 Arbitrage}

An arbitrage is an opportunity to make risk-free profits in the market.
Basic modelling assumption is that no such opportunity exists.
\[
\begin{aligned}
(\bar S_n)_{0\leq n\leq T},\qquad
\bar S_n&=(S_n^0,S_n),& S_n&=(S_n^1,\ldots,S_n^d)
&&\text{asset price model},\\
(\bar\Theta_n)_{1\leq n\leq T},\qquad
\bar\Theta_n&=(\Theta_n^0,\Theta_n),&
\Theta_n&=(\Theta_n^1,\ldots,\Theta_n^d)
&&\text{portfolio}.
\end{aligned}
\]
Value
\[
V_0=\bar\Theta_1\cdot\bar X_0,\qquad
V_n=\bar\Theta_n\cdot\bar X_n,\quad n\geq1,
\qquad X_n^i=S_n^i/S_n^0.
\]
\[
V_n=V_0+\sum_{k=1}^n\Theta_k\cdot(X_k-X_{k-1}).
\]
In this set-up an \emph{arbitrage} for $(\bar S_n)_{0\leq n\leq T}$
is a previsible self-financing portfolio with initial value $0$ such that
\[
V_T\geq0\quad\text{a.s.},\qquad
V_T>0\quad\text{with positive probability}.
\]

Since $V_T$ depends on $(\bar S_n)_{0\leq n\leq T}$ and
$(\bar\Theta_n)_{1\leq n\leq T}$ only through
\[
V_0\text{ and }(X_n)_{0\leq n\leq T}\text{ and }(\Theta_n)_{1\leq n\leq T},
\]
we will often say $(\Theta_n)_{1\leq n\leq T}$ is an arbitrage for
$(X_n)_{0\leq n\leq T}$.

Consider the case $(X_n)_{0\leq n\leq T}$ is a martingale in $\mathbb R^d$
(all components are mgs) and let $(\Theta_n)_{1\leq n\leq T}$ be a bounded
previsible process in $\mathbb R^d$, considered as a portfolio.
Then
\[
V_n=\sum_{k=1}^n\Theta_k\cdot(X_k-X_{k-1})
\qquad\text{is the martingale transform}
\]
\[
\left(=\sum_{i=1}^d\sum_{k=1}^n\Theta_k^i(X_k^i-X_{k-1}^i)\right).
\]
Suppose $\Theta$ chosen so that $V_T\geq0$ a.s.
We have $\mathbb E(V_T)=0$ so $V_T=0$ a.s.

So we see that $(X_n)_{0\leq n\leq T}$ has no bounded arbitrage.

\medskip
\noindent\textbf{Proposition.}
Let $(X_n)_{0\leq n\leq T}$ be a martingale in $\mathbb R^d$.
Then $(X_n)_{0\leq n\leq T}$ has no arbitrage.

\medskip
\noindent\textbf{Proof.}
Let $(\Theta_n)_{1\leq n\leq T}$ be any previsible process. Let
\[
V_n=\sum_{k=1}^n\Theta_k\cdot(X_k-X_{k-1}).
\]
Suppose $V_T\geq0$ (WTS $V_T=0$ a.s.).
Note that
\[
\mathbb E(V_T\mid\mathcal F_T)=V_T\quad\text{a.s.}
\]
Suppose in a backwards induction for $n\leq T$ that
\[
\mathbb E(V_T\mid\mathcal F_n)=V_n\quad\text{a.s.}
\]
Note that then $V_n\geq0$ a.s.
Now
\[
V_n=V_{n-1}+\Theta_n\cdot(X_n-X_{n-1}).
\]
Fix $R<\infty$, set
\[
A=\{|V_{n-1}|\leq R\text{ and }|\Theta_n|\leq R\}.
\]
Then $A$ is $\mathcal F_{n-1}$-meas.\ and
\[
\mathbf1_AV_n=\mathbf1_AV_{n-1}
+(\Theta_n\mathbf1_A)\cdot(X_n-X_{n-1}).
\]
Now take $\mathbb E(\,\cdot\mid\mathcal F_{n-1})$, taking out what is known,
\[
\mathbf1_A\mathbb E(V_n\mid\mathcal F_{n-1})
=\mathbf1_AV_{n-1}+(\Theta_n\mathbf1_A)\,
\underbrace{\mathbb E(X_n-X_{n-1}\mid\mathcal F_{n-1})}_{0}.
\]
But $R$ was arbitrary and $|\Theta_n|$, $|V_{n-1}|$ are finite-valued,
so we must have
\[
\mathbb E(V_n\mid\mathcal F_{n-1})=V_{n-1}\quad\text{a.s.}
\]
so, by tower,
\[
\mathbb E(V_T\mid\mathcal F_{n-1})=V_{n-1}\quad\text{a.s.}
\]
Hence $\mathbb E(V_T)=V_0=0$ so $V_T=0$ a.s., so
$(\Theta_n)_{1\leq n\leq T}$ is not an arbitrage.
\hfill$\square$

\subsection*{2.5 Fundamental Theorem of Asset Pricing}

\noindent\textbf{Theorem.}
For any asset price model $(\bar S_n)_{0\leq n\leq T}$ the following are equivalent:
\begin{enumerate}
    \item[(a)] $(\bar S_n)_{0\leq n\leq T}$ has no arbitrage.
    \item[(b)] There exists an equivalent probability measure $\widetilde{\mathbb P}$
    under which the discounted price process $(X_n)_{0\leq n\leq T}$
    is a martingale.
\end{enumerate}
\[
\widetilde{\mathbb P}\text{ is called an }
\left\{\begin{array}{l}
\text{equivalent martingale}\\
\text{risk-neutral}
\end{array}\right\}\text{ measure}.
\]

\subsubsection*{Recap of the basic framework}

Let $(\bar S_n)_{0\leq n\leq T}$ be an asset price model in
$\mathbb R^{d+1}$, with numeraire $S_n^0>0$, and put
\[
X_n=\frac{S_n}{S_n^0},\qquad \bar X_n=(1,X_n).
\]
Given an initial discounted wealth $V_0$ and a previsible process
$(\Theta_n)_{1\leq n\leq T}$ in $\mathbb R^d$, there is an associated
self-financing portfolio $(\bar\Theta_n)_{1\leq n\leq T}$ with
\[
V_T=\bar\Theta_T\cdot\bar X_T
=V_0+\sum_{n=1}^T\Theta_n\cdot(X_n-X_{n-1}).
\]
Recall that $\Theta$ is an arbitrage if
\[
V_0=0,\qquad V_T\geq0\ \text{a.s.},\qquad
\mathbb P(V_T>0)>0.
\]
An \emph{equivalent martingale measure} is a probability measure
$\widetilde{\mathbb P}\sim\mathbb P$ under which $X$ is a martingale.

\subsubsection*{Proof of the Fundamental Theorem of Asset Pricing}

The theorem says that the following are equivalent:
\begin{enumerate}
    \item[(a)] $X$ has no arbitrage.
    \item[(b)] There exists an equivalent martingale measure.
\end{enumerate}

\noindent\textbf{Proof of (b) $\Rightarrow$ (a).}
By the proposition in Section 2.4, a martingale price process has no
arbitrage under $\widetilde{\mathbb P}$. Equivalence means that
$\mathbb P$ and $\widetilde{\mathbb P}$ have the same null events, so the
conditions defining an arbitrage are the same under either measure.
Hence there is no arbitrage under $\mathbb P$.
\hfill$\square$

\medskip
\noindent\textbf{Proof of (a) $\Rightarrow$ (b) in the single-period case.}
We prove this direction here when $T=1$ and
$\mathcal F_0=\{\varnothing,\Omega\}$.
Write
\[
\Theta=\Theta_1\in\mathbb R^d,\qquad Y=X_1-X_0.
\]
Since $\mathcal F_0$ is trivial, $X_0$ and $\Theta_1$ are deterministic
(up to null sets). Thus (a) says that there is no $\Theta\in\mathbb R^d$
such that
\[
\Theta\cdot Y\geq0\ \text{a.s.},\qquad
\mathbb P(\Theta\cdot Y>0)>0.
\]
Condition (b) asks for $\widetilde{\mathbb P}\sim\mathbb P$ such that
$Y$ is integrable and
\[
\widetilde{\mathbb E}(Y)=0.
\]
Indeed, this is precisely the martingale condition
$\widetilde{\mathbb E}(X_1\mid\mathcal F_0)=X_0$ in this setting.

\paragraph{Step 1: ensure finite exponential moments.}
Set
\[
\phi(\Theta)=\mathbb E\!\left(e^{\Theta\cdot Y}\right),
\qquad \Theta\in\mathbb R^d.
\]
We may reduce to the case $\phi(\Theta)<\infty$ for every $\Theta$ by
first switching to the equivalent probability measure
$\widehat{\mathbb P}$ given by
\[
\frac{d\widehat{\mathbb P}}{d\mathbb P}
=\frac{e^{-|Y|^2}}{\mathbb E(e^{-|Y|^2})}.
\]
The denominator is strictly positive and finite, and the density is
strictly positive. Moreover,
\[
\Theta\cdot Y-|Y|^2
=-\left|Y-\frac{\Theta}{2}\right|^2+\frac{|\Theta|^2}{4}
\leq\frac{|\Theta|^2}{4},
\]
so
\[
\widehat{\mathbb E}\!\left(e^{\Theta\cdot Y}\right)
\leq\frac{e^{|\Theta|^2/4}}{\mathbb E(e^{-|Y|^2})}<\infty.
\]
No arbitrage is preserved under this change of measure. A measure
equivalent to $\widehat{\mathbb P}$ will also be equivalent to the
original $\mathbb P$. We therefore relabel $\widehat{\mathbb P}$ as
$\mathbb P$ for the remainder of the proof and assume all exponential
moments are finite.

Under this assumption $Y$ is integrable and $\phi$ is differentiable,
with
\[
\nabla\phi(\Theta)
=\mathbb E\!\left(Y e^{\Theta\cdot Y}\right).
\]
To justify differentiation under the expectation, on a bounded set of
$\Theta$ the derivatives are dominated by an integrable multiple of
$e^{a\sum_{i=1}^d|Y^i|}$ for a sufficiently large $a$. This is integrable
because
\[
e^{a\sum_{i=1}^d|Y^i|}
\leq\sum_{s\in\{-1,1\}^d}e^{a s\cdot Y},
\]
and every term has finite expectation.

\paragraph{Step 2: reduce the existence of a minimum to relevant directions.}
We will show that $\phi$ attains a minimum on $\mathbb R^d$.
Let
\[
E_0=\{\Theta\in\mathbb R^d:\Theta\cdot Y=0\ \text{a.s.}\},
\qquad E_1=E_0^\perp,
\qquad \mathbb R^d=E_0\oplus E_1.
\]
For $\Theta_0\in E_0$ and $\Theta_1\in E_1$,
\[
\phi(\Theta_0+\Theta_1)=\phi(\Theta_1).
\]
Thus it suffices to find a minimum on $E_1$. If $E_1=\{0\}$, then
$\phi\equiv1$ and the minimum is already attained. Otherwise, since
$\phi(0)=1$, it will suffice to show that
\[
\phi(\Theta)>1
\qquad\text{for all }\Theta\in E_1\text{ with }|\Theta|
\text{ sufficiently large}.
\]

Define
\[
\psi(t)=\min\{1,\max\{0,t\}\},\qquad
f(\Theta)=\mathbb E\!\left(\psi(\Theta\cdot Y)\right).
\]
The function $\psi$ is Lipschitz with constant $1$. Also, $f$ is
continuous by bounded convergence, since $0\leq\psi\leq1$.

Let
\[
\mathcal S=\{\Theta\in E_1:|\Theta|=1\}.
\]
For $\Theta\in\mathcal S$ we have
$\mathbb P(\Theta\cdot Y\neq0)>0$, since
$E_0\cap E_1=\{0\}$. No arbitrage, applied to $-\Theta$, implies
\[
\mathbb P(\Theta\cdot Y>0)>0.
\]
Otherwise $\Theta\cdot Y\leq0$ a.s., and $-\Theta$ would give a
non-negative gain which is strictly positive with positive probability.
It follows that $f(\Theta)>0$ on $\mathcal S$.

Since $\mathcal S$ is compact and $f$ is continuous, set
\[
\varepsilon=\frac12\min_{\Theta\in\mathcal S}f(\Theta)>0.
\]
For every $\Theta\in\mathcal S$, using the Lipschitz property of $\psi$,
\[
\begin{aligned}
\mathbb P(\Theta\cdot Y\geq\varepsilon)
&\geq\mathbb E\!\left(\psi(\Theta\cdot Y-\varepsilon)\right)\\
&\geq\mathbb E\!\left(\psi(\Theta\cdot Y)\right)-\varepsilon\\
&=f(\Theta)-\varepsilon\geq\varepsilon.
\end{aligned}
\]
The first inequality holds because
$\psi(z-\varepsilon)\leq\mathbf1_{\{z\geq\varepsilon\}}$.
Consequently, for $t\geq0$,
\[
\begin{aligned}
\phi(t\Theta)
&=\mathbb E\!\left(e^{t\Theta\cdot Y}\right)\\
&\geq e^{t\varepsilon}
   \mathbb P(\Theta\cdot Y\geq\varepsilon)
\geq\varepsilon e^{t\varepsilon}
\longrightarrow\infty\qquad(t\to\infty).
\end{aligned}
\]
The lower bound is uniform over $\Theta\in\mathcal S$. Writing any
non-zero $\Theta\in E_1$ as $|\Theta|(\Theta/|\Theta|)$ therefore shows
that $\phi(\Theta)>1$ outside a sufficiently large ball in $E_1$.
By continuity, $\phi$ attains a minimum on the compact ball, hence on
$E_1$ and on $\mathbb R^d$. Denote a minimizer by $\Theta^*$.

\paragraph{Step 3: construct the equivalent martingale measure.}
Set
\[
Z=\phi(\Theta^*)\in(0,\infty),\qquad
\frac{d\widetilde{\mathbb P}}{d\mathbb P}
=\frac{e^{\Theta^*\cdot Y}}{Z}.
\]
This is a strictly positive density with expectation $1$, so
$\widetilde{\mathbb P}\sim\mathbb P$.
Since $\Theta^*$ minimizes the differentiable function $\phi$,
\[
\nabla\phi(\Theta^*)
=\mathbb E\!\left(Y e^{\Theta^*\cdot Y}\right)=0.
\]
The same exponential-moment bounds used above give
\[
\widetilde{\mathbb E}(|Y|)
=\frac{\mathbb E\!\left(|Y|e^{\Theta^*\cdot Y}\right)}{Z}
<\infty,
\qquad
\widetilde{\mathbb E}(Y)
=\frac{\nabla\phi(\Theta^*)}{Z}=0.
\]
For example, coordinatewise integrability follows from
$|x|\leq e^x+e^{-x}$ and the finiteness of the exponential moments at
$\Theta^*+e_i$ and $\Theta^*-e_i$.
Thus $X$ is a martingale under $\widetilde{\mathbb P}$, proving (b)
in the single-period case.
\hfill$\square$

\subsection*{2.6 Pricing attainable claims}

Let $C$ be a time-$T$ contingent claim: $C$ is non-negative and
$\mathcal F_T$-measurable. Its \emph{discounted value} is
\[
D=\frac{C}{S_T^0}.
\]
We say that $C$ is \emph{attainable} if there is a previsible
self-financing portfolio $(\bar\Theta_n)_{1\leq n\leq T}$ such that
\[
C=\bar\Theta_T\cdot\bar S_T.
\]
Equivalently, there exist an $\mathcal F_0$-measurable initial
discounted wealth $V_0$ and a previsible process
$(\Theta_n)_{1\leq n\leq T}$ in $\mathbb R^d$ such that
\[
D=V_0+\sum_{n=1}^T\Theta_n\cdot(X_n-X_{n-1}).
\]
Then $V_0$ is called the \emph{fair price} of $C$ in discounted units,
and $\bar\Theta$ is called a \emph{replicating} or \emph{hedging
portfolio}. In currency units the initial price is $S_0^0V_0$.

\medskip
\noindent\textbf{Proposition.}
Assume that
\[
\mathcal F_0=\{\varnothing,\Omega\},\qquad
\mathcal F_T=\sigma(\bar S_1,\ldots,\bar S_T).
\]
Let $C$ be an attainable contingent claim and let
$\widetilde{\mathbb P}$ be any equivalent martingale measure. Then
\[
\boxed{\displaystyle
V_0=\widetilde{\mathbb E}\!\left(\frac{C}{S_T^0}\right).}
\]

\noindent\textbf{Proof.}
Since $C$ is attainable, there is a previsible process $\Theta$ with
\[
D=\frac{C}{S_T^0}
=V_0+\sum_{n=1}^T\Theta_n\cdot(X_n-X_{n-1}).
\]
Since $D\geq0$, the backwards conditional-expectation argument used in
the proposition in Section 2.4 gives
\[
\widetilde{\mathbb E}(D\mid\mathcal F_n)=V_n,
\qquad 0\leq n\leq T.
\]
Here non-negativity of the terminal value allows the same localization
argument even if the previsible holdings are unbounded. At time $0$,
$V_0$ is deterministic, so
\[
\widetilde{\mathbb E}(D)=V_0.
\]
In particular $D$ is integrable under $\widetilde{\mathbb P}$.
\hfill$\square$

\medskip
\noindent\textbf{Interpretation.}
For an attainable claim, every equivalent martingale measure gives the
same discounted expected payoff: it equals the initial cost of a
replicating portfolio.
